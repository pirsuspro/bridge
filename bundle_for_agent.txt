#  Z.ai Bridge Agent  v2.3.0
#  Связывает локальную папку с песочницей Z.ai.
#  Работает на Windows PowerShell 5.1+ (и pwsh 7+), без зависимостей.
#
#  Возможности v2 (сверх v1):
#   - exec: запуск команд с выводом (stdout/stderr/код возврата)
#   - фоновые процессы: exec_start / proc_read / proc_write / proc_kill / proc_list
#   - read_text / write_text: быстрые текстовые правки файлов
#   - restart: перезапуск агента (подхватится bat-лаунчером)
#
#  v2.1.0: постоянный clientId — перезапуск моста больше не вызывает конфликт 409
#          на сервере; 409 обрабатывается мягко (ожидание, а не ошибка).
#  v2.2.0: умная кодировка вывода: сырые байты + эвристика (строгий UTF-8,
#          иначе CP866) — кириллица из cmd, PowerShell и сборщиков читается
#          корректно; stdin пишется в UTF-8; Python-дети принудительно UTF-8.
#
#  v2.3.0: скорость передач (уроки motty — десктопного SSH-клиента:
#          постоянные соединения, бинарный фрейминг, сжатие):
#   - чанки ПК<->песочница — сырые байты (application/octet-stream)
#     вместо base64-in-JSON (−25..33% трафика, без ConvertTo-Json на 1.4 МБ);
#   - gzip чанков в обе стороны, когда выгодно (текст/код — в 3-5 раз меньше);
#   - Expect100Continue/Nagle выкл, ConnectionLimit 8 — отклик poll/чанков
#     без лишних RTT (мелкие POST больше не ждут буферизации Nagle 200 мс);

# ============================================================

function Read-FolderFromUser {
    while ($true) {
        Write-Host ''
        Write-Host '  Укажите папку на этом компьютере, к которой песочница получит доступ.'
        Write-Host ('  Например: C:\Users\' + $env:USERNAME + '\Documents\bridge-share')
        $raw = Read-Host 'Путь к папке'
        $candidate = $raw.Trim().Trim('"').Trim()
        if (-not $candidate) { continue }
        $full = $null
        try { $full = [IO.Path]::GetFullPath($candidate) } catch {
            Write-Host ("  Некорректный путь: " + $candidate); continue
        }
        if (Test-Path -LiteralPath $full -PathType Container) { return $full }
        if (Test-Path -LiteralPath $full) {
            Write-Host ("  Это файл, а не папка: " + $full); continue
        }
        $ans = Read-Host ("  Папка не существует: " + $full + ". Создать? (y - да)")
        if ($ans -match '^[YyDd]') {
            try {
                New-Item -ItemType Directory -Path $full -Force | Out-Null
                if (Test-Path -LiteralPath $full -PathType Container) { return $full }
            } catch { Write-Host ("  Не удалось создать: " + $_.Exception.Message) }
        }
    }
}

function Get-HttpErrorMessage {
    param($Record)
    try {
        $ex = $Record.Exception
        if ($ex -and $ex.Response) {
            $code = ''
            try { $code = [int]$ex.Response.StatusCode } catch { $code = '' }
            $bodyText = ''
            try {
                $stream = $null
                try { $stream = $ex.Response.GetResponseStream() } catch { $stream = $null }
                if ($stream) {
                    $sr = New-Object System.IO.StreamReader($stream, [System.Text.Encoding]::UTF8)
                    $bodyText = $sr.ReadToEnd()
                }
            } catch { }
            if ($bodyText) { return ("HTTP " + $code + " " + $bodyText) }
            return ("HTTP " + $code)
        }
    } catch { }
    return $Record.Exception.Message
}

function Invoke-ApiOnce {
    param([string]$Method, [string]$Path, $BodyObj, [byte[]]$BodyBytes, [string]$ContentType = 'application/octet-stream', [int]$TimeoutSec = 60)
    $uri = $ApiBase + $Path
    $splat = @{
        Uri            = $uri
        Method         = $Method
        Headers        = @{ 'x-bridge-token' = $Token }
        TimeoutSec     = $TimeoutSec
        UseBasicParsing = $true
    }
    if ($null -ne $BodyBytes) {
        # v2.3: бинарное тело (чанк файла, возможно gzip)
        $splat.Body = $BodyBytes
        $splat.ContentType = $ContentType
    } elseif ($null -ne $BodyObj) {
        $json = ConvertTo-Json -InputObject $BodyObj -Depth 10 -Compress
        $splat.Body = [System.Text.Encoding]::UTF8.GetBytes($json)
        $splat.ContentType = 'application/json; charset=utf-8'
    }
    $resp = Invoke-WebRequest @splat
    $text = ''
    try {
        if ($resp.RawContentStream) {
            $resp.RawContentStream.Position = 0
            $sr = New-Object System.IO.StreamReader($resp.RawContentStream, [System.Text.Encoding]::UTF8)
            $text = $sr.ReadToEnd()
            $sr.Close()
        }
    } catch { $text = '' }
    if ([string]::IsNullOrWhiteSpace($text)) { return $null }
    return ($text | ConvertFrom-Json)
}

# v2.3: бинарный GET (чанк файла): сырой ответ + заголовки x-index/x-last/x-eof/x-gz
function Invoke-ApiRawResponse {
    param([string]$Method, [string]$Path, [int]$Retries = 4, [int]$TimeoutSec = 60)
    $attempt = 0
    while ($true) {
        try {
            $uri = $ApiBase + $Path
            return Invoke-WebRequest -Uri $uri -Method $Method `
                -Headers @{ 'x-bridge-token' = $Token } `
                -TimeoutSec $TimeoutSec -UseBasicParsing
        } catch {
            $attempt++
            $msg = Get-HttpErrorMessage -Record $_
            if ($attempt -gt $Retries) { throw ("Сетевая ошибка (" + $Path + "): " + $msg) }
            if ($msg -like 'HTTP 4*' -and $msg -notlike 'HTTP 429*') {
                throw ("Ошибка API (" + $Path + "): " + $msg)
            }
            Start-Sleep -Milliseconds (400 * $attempt)
        }
    }
}

# v2.3: заголовок ответа без учёта регистра (PS 5.1 Headers — регистрозависим)
function Get-RespHeader {
    param($Response, [string]$Name)
    if ($null -eq $Response -or $null -eq $Response.Headers) { return '' }
    foreach ($k in @($Response.Headers.Keys)) {
        if ([string]::Compare([string]$k, $Name, $true) -eq 0) {
            return [string]$Response.Headers[$k]
        }
    }
    return ''
}

# v2.3: gzip-сжатие байтов (чанк ПК -> песочница)
function Compress-Bytes {
    param([byte[]]$Data)
    $ms = New-Object System.IO.MemoryStream
    $gz = New-Object System.IO.Compression.GZipStream($ms, [System.IO.Compression.CompressionLevel]::Fastest, $true)
    $gz.Write($Data, 0, $Data.Length)
    $gz.Close()
    $out = $ms.ToArray()
    $ms.Close()
    return $out
}

# v2.3: gzip-распаковка байтов (чанк песочница -> ПК)
function Expand-Bytes {
    param([byte[]]$Data)
    $in = New-Object System.IO.MemoryStream(, $Data)
    $gz = New-Object System.IO.Compression.GZipStream($in, [System.IO.Compression.CompressionMode]::Decompress)
    $out = New-Object System.IO.MemoryStream
    $gz.CopyTo($out)
    $gz.Close()
    $in.Close()
    $bytes = $out.ToArray()
    $out.Close()
    return $bytes
}

function Invoke-Api {
    param([string]$Method, [string]$Path, $BodyObj, [byte[]]$BodyBytes, [string]$ContentType = 'application/octet-stream', [int]$Retries = 4, [int]$TimeoutSec = 60)
    $attempt = 0
    while ($true) {

            return Invoke-ApiOnce -Method $Method -Path $Path -BodyObj $BodyObj -BodyBytes $BodyBytes -ContentType $ContentType -TimeoutSec $TimeoutSec

            if ($msg -like 'HTTP 4*' -and $msg -notlike 'HTTP 429*') {
                throw ("Ошибка API (" + $Path + "): " + $msg)
            }
            Start-Sleep -Milliseconds (400 * $attempt)
        }
    }
}

function Send-Result {
    param([string]$Id, [bool]$Ok, $Data, [string]$ErrorMessage)
    $body = @{ id = $Id; ok = $Ok }
    if ($Ok) {
        if ($null -ne $Data) { $body.data = $Data }
    } else {
        $body.error = $ErrorMessage
    }
    Invoke-Api -Method 'Post' -Path '/result' -BodyObj $body -Retries 6 | Out-Null
}

# ============ Безопасные пути (только внутри выбранной папки) ============

function Resolve-BridgePath {
    param([string]$RelPath)
    if ($null -eq $RelPath) { $RelPath = '' }
    $RelPath = ([string]$RelPath).Trim().TrimStart('\', '/')
    if ($RelPath -eq '') { return $script:Root }
    $parts = @($RelPath -split '[\\/]' | Where-Object { $_ -ne '' -and $_ -ne '.' })
    if ($parts -contains '..') { throw ("Путь не должен содержать '..': " + $RelPath) }
    $cur = $script:Root
    foreach ($p in $parts) { $cur = [IO.Path]::Combine($cur, $p) }
    $full = [IO.Path]::GetFullPath($cur)
    $rootFull = [IO.Path]::GetFullPath($script:Root)
    $rootPrefix = $rootFull.TrimEnd($Sep) + $Sep
    $cmpType = if ($IsWin) { [StringComparison]::OrdinalIgnoreCase } else { [StringComparison]::Ordinal }
    if (-not $full.StartsWith($rootPrefix, $cmpType) -and $full -ne $rootFull.TrimEnd($Sep)) {
        throw ("Путь вне папки моста: " + $RelPath)
    }
    return $full
}

function Get-RelPath {
    param([string]$FullPath)
    $rootFull = [IO.Path]::GetFullPath($script:Root)
    $rootPrefix = $rootFull.TrimEnd($Sep) + $Sep
    $rel = $FullPath.Substring($rootPrefix.Length)
    return ($rel -replace '\\', '/')
}

# ============ v2: движок процессов (C#, потокобезопасно) ============

$script:ExecAvailable = $false
try {
    Add-Type -TypeDefinition @"
using System;
using System.Collections.Generic;
using System.Diagnostics;
using System.Text;
using System.Threading;

namespace ZaiBridge
{
    public class ProcEntry
    {
        public string Handle;
        public Process Proc;
        public int Pid;
        public string Cmd;
        public string Cwd;
        public DateTime StartedAt;
        public long DurationMs;
        public int ExitCode;
        public bool ExitKnown;
        public bool TimedOut;
        public bool Killed;
        public StringBuilder Out = new StringBuilder(65536);
        public StringBuilder Err = new StringBuilder(4096);
        public object Gate = new object();
    }

    public static class ProcMan
    {
        private static readonly Dictionary<string, ProcEntry> Procs = new Dictionary<string, ProcEntry>();
        private static readonly object LockObj = new object();
        private static long _seq = 0;

        private static ProcEntry Get(string handle)
        {
            lock (LockObj)
            {
                ProcEntry entry;
                if (!Procs.TryGetValue(handle, out entry))
                {
                    throw new Exception("process not found: " + handle);
                }
                return entry;
            }
        }

        // Latin-1 (28591): каждый байт дочернего процесса ложится в char 1:1,
        // настоящая кодировка определяется при чтении (ReadDecoded).
        private static readonly Encoding RawEnc = Encoding.GetEncoding(28591);
        private static readonly UTF8Encoding StrictUtf8 = new UTF8Encoding(false, true);

        public static string Start(string cmd, string cwd, Dictionary<string,string> env)
        {
            ProcessStartInfo psi = new ProcessStartInfo();
            psi.FileName = "cmd.exe";
            psi.Arguments = "/d /c chcp 65001>nul & " + cmd;
            psi.UseShellExecute = false;
            psi.CreateNoWindow = true;
            psi.RedirectStandardOutput = true;
            psi.RedirectStandardError = true;
            psi.RedirectStandardInput = true;
            psi.StandardOutputEncoding = RawEnc;
            psi.StandardErrorEncoding = RawEnc;
            if (!String.IsNullOrEmpty(cwd)) { psi.WorkingDirectory = cwd; }
            // Подталкиваем Python-инструменты к UTF-8 (остальным не мешает)
            psi.EnvironmentVariables["PYTHONUTF8"] = "1";
            psi.EnvironmentVariables["PYTHONIOENCODING"] = "utf-8";
            if (env != null)
            {
                foreach (KeyValuePair<string,string> kv in env)
                {
                    psi.EnvironmentVariables[kv.Key] = kv.Value;
                }
            }
            Process p = new Process();
            p.StartInfo = psi;
            p.EnableRaisingEvents = true;
            ProcEntry entry = new ProcEntry();
            entry.Cmd = cmd;
            entry.Cwd = (cwd == null ? "" : cwd);
            entry.Handle = "p" + Interlocked.Increment(ref _seq).ToString("x");
            entry.StartedAt = DateTime.Now;
            p.OutputDataReceived += delegate(object s, DataReceivedEventArgs ev)
            {
                if (ev.Data != null) { lock (entry.Gate) { entry.Out.AppendLine(ev.Data); } }
            };
            p.ErrorDataReceived += delegate(object s, DataReceivedEventArgs ev)
            {
                if (ev.Data != null) { lock (entry.Gate) { entry.Err.AppendLine(ev.Data); } }
            };
            p.Start();
            entry.Proc = p;
            entry.Pid = p.Id;
            p.BeginOutputReadLine();
            p.BeginErrorReadLine();
            lock (LockObj) { Procs[entry.Handle] = entry; }
            return entry.Handle;
        }

        public static bool HasExited(string handle)
        {
            ProcEntry entry = Get(handle);
            lock (LockObj)
            {
                if (entry.ExitKnown) { return true; }
                if (entry.Proc.HasExited)
                {
                    entry.ExitKnown = true;
                    try { entry.ExitCode = entry.Proc.ExitCode; }
                    catch (Exception) { entry.ExitCode = -1; }
                    entry.DurationMs = (long)(DateTime.Now - entry.StartedAt).TotalMilliseconds;
                    return true;
                }
                return false;
            }
        }

        public static int GetPid(string handle) { return Get(handle).Pid; }
        public static string GetCmd(string handle) { return Get(handle).Cmd; }
        public static long GetDurationMs(string handle) { return Get(handle).DurationMs; }
        public static int GetExitCode(string handle) { return Get(handle).ExitCode; }
        public static bool IsTimedOut(string handle) { return Get(handle).TimedOut; }

        public static DateTime GetStartedAt(string handle) { return Get(handle).StartedAt; }

        public static void SetTimedOut(string handle)
        {
            lock (LockObj) { Get(handle).TimedOut = true; }
        }

        public static string Read(string handle, int stream, int from, int maxChars, out int next)
        {
            ProcEntry entry = Get(handle);
            StringBuilder sb = (stream == 0 ? entry.Out : entry.Err);
            lock (entry.Gate)
            {
                if (from < 0) { from = 0; }
                if (from > sb.Length) { from = sb.Length; }
                int len = sb.Length - from;
                if (len > maxChars) { len = maxChars; }
                next = from + len;
                return sb.ToString(from, len);
            }
        }

        // Эвристика кодировки вывода. Windows-утилиты пишут в pipe по-разному:
        // cmd.exe — CP866 (системная OEM), PowerShell/Python/сборщики — UTF-8.
        // Шаг 1: весь чанк валиден как UTF-8 — это он и есть.
        // Шаг 2: построчно (cmd-эхо вперемешку с UTF-8-выводом инструментов).
        public static string ReadDecoded(string handle, int stream, int from, int maxChars, out int next, out string encName)
        {
            string raw = Read(handle, stream, from, maxChars, out next);
            if (raw.Length == 0) { encName = ""; return ""; }
            byte[] bytes = new byte[raw.Length];
            for (int i = 0; i < raw.Length; i++) { bytes[i] = (byte)raw[i]; }
            try
            {
                string whole = StrictUtf8.GetString(bytes);
                encName = "utf-8";
                return whole;
            }
            catch (Exception) { }
            string[] lines = raw.Split('\n');
            StringBuilder sb2 = new StringBuilder(raw.Length + 16);
            for (int i = 0; i < lines.Length; i++)
            {
                string ln = lines[i];
                if (ln.Length > 0 && ln[ln.Length - 1] == '\r') { ln = ln.Substring(0, ln.Length - 1); }
                byte[] lb = new byte[ln.Length];
                for (int j = 0; j < ln.Length; j++) { lb[j] = (byte)ln[j]; }
                try { sb2.Append(StrictUtf8.GetString(lb)); }
                catch (Exception) { sb2.Append(Encoding.GetEncoding(866).GetString(lb)); }
                if (i < lines.Length - 1) { sb2.Append("\r\n"); }
            }
            encName = "mixed:cp866/utf-8";
            return sb2.ToString();
        }

        public static void WriteStdin(string handle, string text, bool newLine)
        {
            ProcEntry entry = Get(handle);
            if (!entry.Proc.StartInfo.RedirectStandardInput)
            {
                throw new Exception("stdin is not redirected for this process");
            }
            byte[] bytes = Encoding.UTF8.GetBytes(newLine ? text + "\r\n" : text);
            entry.Proc.StandardInput.BaseStream.Write(bytes, 0, bytes.Length);
            entry.Proc.StandardInput.BaseStream.Flush();
        }

        public static void Kill(string handle)
        {
            ProcEntry entry = Get(handle);
            lock (LockObj)
            {
                entry.Killed = true;
                try
                {
                    ProcessStartInfo psi = new ProcessStartInfo("taskkill", "/PID " + entry.Pid + " /T /F");
                    psi.UseShellExecute = false;
                    psi.CreateNoWindow = true;
                    Process tk = Process.Start(psi);
                    if (tk != null && !tk.WaitForExit(8000)) { tk.Kill(); }
                }
                catch (Exception) { }
                try { if (!entry.Proc.HasExited) { entry.Proc.Kill(); } }
                catch (Exception) { }
            }
        }

        public static string[] Handles()
        {
            lock (LockObj)
            {
                string[] arr = new string[Procs.Keys.Count];
                Procs.Keys.CopyTo(arr, 0);
                return arr;
            }
        }
    }
}
"@
    $script:ExecAvailable = $true
} catch {
    Write-Log ("ВНИМАНИЕ: exec-движок не скомпилировался — файловые команды работают, exec нет. Причина: " + $_.Exception.Message)
}

# ============ v2: защита от катастрофических команд ============

$DenyPatterns = @(
    '\bformat\s+[a-z]:',
    '\bshutdown\b',
    '\bdiskpart\b',
    '\bmkfs\.',
    '\bbcdedit\b',
    'vssadmin\s+delete'
)

function Test-Denied {
    param([string]$CmdText)
    foreach ($p in $DenyPatterns) {
        if ($CmdText -match $p) { return $true }
    }
    return $false
}

# ============ v2: обработчики запуска команд ============

function ConvertTo-EnvDictionary {
    param($EnvObj)
    if ($null -eq $EnvObj) { return $null }
    $d = New-Object 'System.Collections.Generic.Dictionary[string,string]'
    foreach ($p in $EnvObj.PSObject.Properties) {
        $d.Add([string]$p.Name, [string]$p.Value)
    }
    return $d
}

function Resolve-ExecCwd {
    param([string]$CwdPath)
    if ([string]::IsNullOrWhiteSpace($CwdPath)) { return $script:Root }
    return Resolve-BridgePath $CwdPath
}

function Start-ExecTask {
    param($cmd, $payload)
    if (-not $script:ExecAvailable) { throw 'exec-движок недоступен (ошибка компиляции при старте моста)' }
    $cmdText = [string]$payload.cmd
    if ([string]::IsNullOrWhiteSpace($cmdText)) { throw 'пустая команда' }
    if (Test-Denied $cmdText) { throw ('команда отклонена защитным фильтром: ' + $cmdText) }
    $cwd = Resolve-ExecCwd ([string]$payload.cwd)
    if (-not (Test-Path -LiteralPath $cwd -PathType Container)) { throw ('рабочая папка не найдена: ' + $cwd) }
    $timeoutSec = [double]0
    if ($payload.timeoutSec) { $timeoutSec = [double]$payload.timeoutSec }
    if ($timeoutSec -lt 5) { $timeoutSec = 120 }
    if ($timeoutSec -gt 600) { $timeoutSec = 600 }
    $envDict = ConvertTo-EnvDictionary $payload.env
    $handle = [ZaiBridge.ProcMan]::Start($cmdText, $cwd, $envDict)
    $task = @{
        handle   = $handle
        deadline = (Get-Date).AddSeconds($timeoutSec + 3)
    }
    $Global:BridgeTasks[[string]$cmd.id] = $task
    Write-Log ("exec [" + $cmd.id + "] запущен (" + $timeoutSec + "с): " + $cmdText)
    return $null
}

function Update-ExecTasks {
    if ($Global:BridgeTasks.Count -eq 0) { return }
    $doneIds = @()
    foreach ($kv in @($Global:BridgeTasks.GetEnumerator())) {
        $id = [string]$kv.Key
        $task = $kv.Value
        $handle = [string]$task.handle
        try {
            $exited = [ZaiBridge.ProcMan]::HasExited($handle)
            if (-not $exited -and ((Get-Date) -gt $task.deadline)) {
                Write-Log ("exec [" + $id + "] таймаут — убиваю дерево процессов")
                [ZaiBridge.ProcMan]::Kill($handle)
                [ZaiBridge.ProcMan]::SetTimedOut($handle)
                Start-Sleep -Milliseconds 600
                $exited = [ZaiBridge.ProcMan]::HasExited($handle)
                if (-not $exited) { continue }
            }
            if ($exited) {
                $nextOut = 0
                $nextErr = 0
                $outEnc = ''
                $errEnc = ''
                $out = [ZaiBridge.ProcMan]::ReadDecoded($handle, 0, 0, 400000, [ref]$nextOut, [ref]$outEnc)
                $err = [ZaiBridge.ProcMan]::ReadDecoded($handle, 1, 0, 400000, [ref]$nextErr, [ref]$errEnc)
                $code = [ZaiBridge.ProcMan]::GetExitCode($handle)
                $dur = [ZaiBridge.ProcMan]::GetDurationMs($handle)
                $timedOut = [bool][ZaiBridge.ProcMan]::IsTimedOut($handle)
                $data = @{
                    code         = $code
                    stdout       = $out
                    stderr       = $err
                    stdoutEnc    = $outEnc
                    stderrEnc    = $errEnc
                    stdoutTrunc  = ($out.Length -ge 400000)
                    stderrTrunc  = ($err.Length -ge 400000)
                    durationMs   = $dur
                    timedOut     = $timedOut
                }
                Send-Result -Id $id -Ok $true -Data $data
                Write-Log ("exec [" + $id + "] завершён, код " + $code + ", " + $dur + " мс")
                $doneIds += $id
            }
        } catch {
            $msg = $_.Exception.Message
            Write-Log ("Ошибка задачи exec [" + $id + "]: " + $msg)
            try { Send-Result -Id $id -Ok $false -ErrorMessage $msg } catch { }
            $doneIds += $id
        }
    }
    foreach ($id in @($doneIds)) { $Global:BridgeTasks.Remove($id) }
}

function Start-BackgroundProc {
    param($payload)
    if (-not $script:ExecAvailable) { throw 'exec-движок недоступен (ошибка компиляции при старте моста)' }
    $cmdText = [string]$payload.cmd
    if ([string]::IsNullOrWhiteSpace($cmdText)) { throw 'пустая команда' }
    if (Test-Denied $cmdText) { throw ('команда отклонена защитным фильтром: ' + $cmdText) }
    $cwd = Resolve-ExecCwd ([string]$payload.cwd)
    if (-not (Test-Path -LiteralPath $cwd -PathType Container)) { throw ('рабочая папка не найдена: ' + $cwd) }
    $envDict = ConvertTo-EnvDictionary $payload.env
    $handle = [ZaiBridge.ProcMan]::Start($cmdText, $cwd, $envDict)
    $procPid = [ZaiBridge.ProcMan]::GetPid($handle)
    Write-Log ("фон " + $handle + " (pid " + $procPid + "): " + $cmdText)
    return @{ handle = $handle; pid = $procPid; cmd = $cmdText }
}

function Read-BackgroundProc {
    param($payload)
    $handle = [string]$payload.handle
    if (-not $handle) { throw 'не указан handle процесса' }
    $fromOut = 0
    if ($payload.fromOut) { $fromOut = [int]$payload.fromOut }
    $fromErr = 0
    if ($payload.fromErr) { $fromErr = [int]$payload.fromErr }
    $maxChars = 100000
    if ($payload.maxChars) { $maxChars = [int]$payload.maxChars }
    $nextOut = 0
    $nextErr = 0
    $outEnc = ''
    $errEnc = ''
    $out = [ZaiBridge.ProcMan]::ReadDecoded($handle, 0, $fromOut, $maxChars, [ref]$nextOut, [ref]$outEnc)
    $err = [ZaiBridge.ProcMan]::ReadDecoded($handle, 1, $fromErr, $maxChars, [ref]$nextErr, [ref]$errEnc)
    $running = -not [ZaiBridge.ProcMan]::HasExited($handle)
    $res = @{
        handle   = $handle
        pid      = [ZaiBridge.ProcMan]::GetPid($handle)
        running  = $running
        stdout   = $out
        stderr   = $err
        stdoutEnc = $outEnc
        stderrEnc = $errEnc
        nextOut  = $nextOut
        nextErr  = $nextErr
    }
    if (-not $running) {
        $res.exitCode = [ZaiBridge.ProcMan]::GetExitCode($handle)
        $res.durationMs = [ZaiBridge.ProcMan]::GetDurationMs($handle)
    }
    return $res
}

function Write-BackgroundProc {
    param($payload)
    $handle = [string]$payload.handle
    if (-not $handle) { throw 'не указан handle процесса' }
    $inputText = [string]$payload.input
    $newLine = $true
    if ($payload.newline -eq $false) { $newLine = $false }
    if ([ZaiBridge.ProcMan]::HasExited($handle)) { throw 'процесс уже завершён' }
    [ZaiBridge.ProcMan]::WriteStdin($handle, $inputText, $newLine)
    return @{ ok = $true }
}

function Kill-BackgroundProc {
    param($payload)
    $handle = [string]$payload.handle
    if (-not $handle) { throw 'не указан handle процесса' }
    $already = $false
    try {
        if ([ZaiBridge.ProcMan]::HasExited($handle)) { $already = $true }
    } catch { }
    if (-not $already) { [ZaiBridge.ProcMan]::Kill($handle) }
    Write-Log ("kill " + $handle + " (уже завершён: " + $already + ")")
    return @{ ok = $true; alreadyFinished = $already }
}

function List-BackgroundProcs {
    if (-not $script:ExecAvailable) { throw 'exec-движок недоступен' }
    $handles = [ZaiBridge.ProcMan]::Handles()
    $arr = @()
    foreach ($h in $handles) {
        $running = -not [ZaiBridge.ProcMan]::HasExited($h)
        $entry = @{
            handle    = $h
            pid       = [ZaiBridge.ProcMan]::GetPid($h)
            cmd       = [ZaiBridge.ProcMan]::GetCmd($h)
            startedAt = ([ZaiBridge.ProcMan]::GetStartedAt($h)).ToString('yyyy-MM-dd HH:mm:ss')
            running   = $running
        }
        if (-not $running) {
            $entry.exitCode = [ZaiBridge.ProcMan]::GetExitCode($h)
            $entry.durationMs = [ZaiBridge.ProcMan]::GetDurationMs($h)
        }
        $arr += ,$entry
    }
    return @{ procs = $arr }
}

function Read-TextFile {
    param($payload)
    $p = Resolve-BridgePath $payload.path
    if (-not (Test-Path -LiteralPath $p -PathType Leaf)) { throw ('Файл не найден: ' + $payload.path) }
    $size = (Get-Item -LiteralPath $p).Length
    if ($size -gt 2000000) { throw ('Файл больше 2 МБ (' + $size + ' байт) — используйте обычную передачу файлов') }
    $bytes = [IO.File]::ReadAllBytes($p)
    $strictUtf8 = New-Object System.Text.UTF8Encoding($false, $true)
    $content = $null
    $encName = 'utf-8'
    try {
        $content = $strictUtf8.GetString($bytes)
    } catch {
        $encName = 'cp1251'
        $content = [System.Text.Encoding]::GetEncoding(1251).GetString($bytes)
    }
    if ($content.Length -gt 0 -and $content[0] -eq [char]0xFEFF) { $content = $content.Substring(1) }
    $maxChars = 200000
    if ($payload.maxChars) { $maxChars = [int]$payload.maxChars }
    $truncated = $false
    if ($content.Length -gt $maxChars) {
        $content = $content.Substring(0, $maxChars)
        $truncated = $true
    }
    return @{ content = $content; truncated = $truncated; sizeBytes = $size; encoding = $encName; path = [string]$payload.path }
}

function Write-TextFileCmd {
    param($payload)
    $p = Resolve-BridgePath $payload.path
    $content = [string]$payload.content
    $append = $false
    if ($payload.append -eq $true) { $append = $true }
    $dir = Split-Path -Parent $p
    if ($dir -and -not (Test-Path -LiteralPath $dir)) {
        New-Item -ItemType Directory -Path $dir -Force | Out-Null
    }
    $existed = Test-Path -LiteralPath $p
    $utf8 = New-Object System.Text.UTF8Encoding($false)
    if ($append) {
        [IO.File]::AppendAllText($p, $content, $utf8)
    } else {
        [IO.File]::WriteAllText($p, $content, $utf8)
    }
    $modeWord = 'записан'
    if ($append) { $modeWord = 'дополнен' }
    Write-Log ("Текст " + $modeWord + ": " + $payload.path + " (" + $content.Length + " симв)")
    return @{ ok = $true; bytes = ([Text.Encoding]::UTF8.GetByteCount($content)); created = (-not $existed); appended = $append }
}

function Invoke-Restart {
    param($cmd)
    Write-Log 'Команда restart: отвечаю и перезапускаюсь через 2 секунды.'
    try {
        Send-Result -Id $cmd.id -Ok $true -Data @{ ok = $true; restarting = $true }
    } catch { }
    $killer = 'Start-Sleep -Seconds 2; Stop-Process -Id ' + $PID + ' -Force'
    Start-Process -FilePath 'powershell.exe' -ArgumentList @('-NoProfile', '-ExecutionPolicy', 'Bypass', '-Command', $killer) -WindowStyle Hidden
    return $null
}

# ============ Обработчики файловых команд (v1) ============

function Invoke-Stat {
    param($payload)
    $p = Resolve-BridgePath $payload.path
    if (-not (Test-Path -LiteralPath $p)) { return @{ exists = $false } }
    $item = Get-Item -LiteralPath $p -Force
    $isDir = [bool]$item.PSIsContainer
    return @{
        exists = $true
        isDir  = $isDir
        size   = if ($isDir) { 0 } else { [int64]$item.Length }
        mtime  = $item.LastWriteTime.ToString('yyyy-MM-dd HH:mm:ss')
    }
}

function Get-Listing {
    param($payload)
    $dir = Resolve-BridgePath $payload.path
    if (-not (Test-Path -LiteralPath $dir -PathType Container)) {
        throw ("Папка не найдена: " + $payload.path)
    }
    $limit = 2000
    if ($payload.recursive) {
        $items = @(Get-ChildItem -LiteralPath $dir -Recurse -Force -ErrorAction SilentlyContinue)
    } else {
        $items = @(Get-ChildItem -LiteralPath $dir -Force -ErrorAction SilentlyContinue |
            Sort-Object @{ Expression = { -not [bool]$_.PSIsContainer } }, Name)
    }
    $entries = @()
    $truncated = $false
    foreach ($i in $items) {
        if ($entries.Count -ge $limit) { $truncated = $true; break }
        $isDir = [bool]$i.PSIsContainer
        $entries += ,@{
            name  = $i.Name
            path  = (Get-RelPath $i.FullName)
            isDir = $isDir
            size  = if ($isDir) { 0 } else { [int64]$i.Length }
            mtime = $i.LastWriteTime.ToString('yyyy-MM-dd HH:mm:ss')
        }
    }
    return @{ path = [string]$payload.path; entries = $entries; truncated = $truncated }
}

function New-Dir {
    param($payload)
    $p = Resolve-BridgePath $payload.path
    if (-not (Test-Path -LiteralPath $p -PathType Container)) {
        New-Item -ItemType Directory -Path $p -Force | Out-Null
    }
    return @{ ok = $true; path = (Get-RelPath $p) }
}

function Remove-BridgeItem {
    param($payload)
    $p = Resolve-BridgePath $payload.path
    $rootFull = [IO.Path]::GetFullPath($script:Root).TrimEnd($Sep)
    if ($p.TrimEnd($Sep) -eq $rootFull) { throw 'Нельзя удалить корневую папку моста' }
    if (-not (Test-Path -LiteralPath $p)) { return @{ ok = $true; note = 'уже удалено' } }
    Remove-Item -LiteralPath $p -Recurse -Force
    return @{ ok = $true }
}

function Move-BridgeItem {
    param($payload)
    $from = Resolve-BridgePath $payload.from
    $to = Resolve-BridgePath $payload.to
    if (-not (Test-Path -LiteralPath $from)) { throw ("Не найдено: " + $payload.from) }
    if (Test-Path -LiteralPath $to -PathType Container) {
        throw ("Папка с таким именем уже существует: " + $payload.to)
    }
    $parent = Split-Path -Parent $to
    if ($parent -and -not (Test-Path -LiteralPath $parent)) {
        New-Item -ItemType Directory -Path $parent -Force | Out-Null
    }
    if (Test-Path -LiteralPath $to -PathType Leaf) { Remove-Item -LiteralPath $to -Force }
    Move-Item -LiteralPath $from -Destination $to -Force
    return @{ ok = $true }
}

# --- Передача файла: ПК -> песочница ---
function Invoke-DownloadTransfer {
    param($cmd, $payload)
    $tid = [string]$payload.transferId
    $p = Resolve-BridgePath $payload.path
    if (-not (Test-Path -LiteralPath $p -PathType Leaf)) {
        $msg = "Файл не найден: $($payload.path)"
        Send-Result -Id $cmd.id -Ok $false -ErrorMessage $msg
        return $null
    }
    $item = Get-Item -LiteralPath $p
    $size = [int64]$item.Length
    if ($size -gt 2000000000) {
        $msg = 'Файл больше 2 ГБ — не поддерживается'
        Send-Result -Id $cmd.id -Ok $false -ErrorMessage $msg
        return $null
    }
    $hash = ''
    try { $hash = (Get-FileHash -LiteralPath $p -Algorithm SHA256).Hash.ToLower() } catch { $hash = '' }

    # Сначала результат команды (размер и хэш), затем чанки
    Send-Result -Id $cmd.id -Ok $true -Data @{ ok = $true; size = $size; sha256 = $hash }

    $fs = $null
    try {
        $fs = [IO.File]::OpenRead($p)
        $buf = New-Object byte[] $ChunkSize
        $index = 0
        $binOk = ($Global:ServerFeats -contains 'binchunk')
        $gzOk = ($Global:ServerFeats -contains 'gzip')
        while ($true) {
            $n = $fs.Read($buf, 0, $ChunkSize)
            if ($n -le 0) { break }
            if ($n -eq $ChunkSize) {
                $chunk = $buf
            } else {
                $chunk = New-Object byte[] $n
                [Array]::Copy($buf, $chunk, $n)
            }
            if ($binOk) {
                # v2.3: сырые байты (+gzip, если выгодно) вместо base64-JSON
                $chunkPayload = $chunk
                $useGz = $false
                if ($gzOk -and $chunk.Length -gt 512) {
                    try {
                        $packed = Compress-Bytes -Data $chunk
                        if ($packed.Length -lt $chunk.Length) {
                            $chunkPayload = $packed
                            $useGz = $true
                        }
                    } catch { $chunkPayload = $chunk; $useGz = $false }
                }
                $q = "/transfer/$tid/chunk?index=$index"
                if ($useGz) { $q = $q + '&gz=1' }
                Invoke-Api -Method 'Post' -Path $q -BodyBytes $chunkPayload -ContentType 'application/octet-stream' -Retries 6 -TimeoutSec 180 | Out-Null
            } else {
                $b64 = [Convert]::ToBase64String($chunk)
                Invoke-Api -Method 'Post' -Path ("/transfer/$tid/chunk") -BodyObj @{ index = $index; dataB64 = $b64 } -Retries 6 -TimeoutSec 120 | Out-Null
            }
            $Global:BridgeLastHeartbeat = Get-Date
            $index++
        }
        Invoke-Api -Method 'Post' -Path ("/transfer/$tid/complete") -BodyObj @{ sha256 = $hash } -Retries 6 -TimeoutSec 60 | Out-Null
        Write-Log ("Отправлен файл: " + $payload.path + " (" + $size + " байт)")
    } catch {
        $msg = $_.Exception.Message
        Fail-Transfer -TransferId $tid -Message $msg
        Write-Log ("Ошибка отправки файла " + $payload.path + ": " + $msg)
    } finally {
        if ($fs) { $fs.Dispose() }
    }
    return $null
}

# --- Передача файла: песочница -> ПК ---
function Invoke-UploadTransfer {
    param($cmd, $payload)
    $tid = [string]$payload.transferId
    $dest = Resolve-BridgePath $payload.path
    if (Test-Path -LiteralPath $dest -PathType Container) {
        $msg = "Папка с таким именем уже существует: $($payload.path)"
        Send-Result -Id $cmd.id -Ok $false -ErrorMessage $msg
        return $null
    }
    $parent = Split-Path -Parent $dest
    if ($parent -and -not (Test-Path -LiteralPath $parent)) {
        New-Item -ItemType Directory -Path $parent -Force | Out-Null
    }
    $tmp = $dest + '.bridgepart'

    Send-Result -Id $cmd.id -Ok $true -Data @{ ok = $true }

    $fs = $null
    try {
        $fs = [IO.File]::Create($tmp)
        $index = 0
        $total = [int64]0
        $binOk = ($Global:ServerFeats -contains 'binchunk')
        $gzOk = ($Global:ServerFeats -contains 'gzip')
        while ($true) {
            if ($binOk) {
                # v2.3: сырой чанк (+gzip) с метаданными в заголовках
                $q = "/transfer/$tid/chunk?index=$index&bin=1"
                if ($gzOk) { $q = $q + '&gz=1' }
                $r = Invoke-ApiRawResponse -Method 'Get' -Path $q -Retries 6 -TimeoutSec 180
                if ((Get-RespHeader -Response $r -Name 'x-eof') -eq '1') { break }
                $ms = New-Object System.IO.MemoryStream
                try {
                    if ($r.RawContentStream) {
                        $r.RawContentStream.Position = 0
                        $r.RawContentStream.CopyTo($ms)
                    }
                } catch { }
                $bytes = $ms.ToArray()
                $ms.Close()
                if ((Get-RespHeader -Response $r -Name 'x-gz') -eq '1') {
                    $bytes = Expand-Bytes -Data $bytes
                }
                if ($bytes.Length -eq 0) { break }
                $fs.Write($bytes, 0, $bytes.Length)
                $total += $bytes.Length
                $Global:BridgeLastHeartbeat = Get-Date
                if ((Get-RespHeader -Response $r -Name 'x-last') -eq '1') { break }
                $index++
            } else {
                $r = Invoke-Api -Method 'Get' -Path ("/transfer/$tid/chunk?index=" + $index) -Retries 6 -TimeoutSec 120
                if ($null -eq $r -or $r.eof) { break }
                $bytes = [Convert]::FromBase64String([string]$r.dataB64)
                $fs.Write($bytes, 0, $bytes.Length)
                $total += $bytes.Length
                $Global:BridgeLastHeartbeat = Get-Date
                if ($r.last) { break }
                $index++
            }
        }
        $fs.Dispose()
        $fs = $null
        $hash = (Get-FileHash -LiteralPath $tmp -Algorithm SHA256).Hash.ToLower()
        $expected = ''
        if ($payload.sha256) { $expected = ([string]$payload.sha256).ToLower() }
        if ($expected -and $hash -ne $expected) {
            Remove-Item -LiteralPath $tmp -Force -ErrorAction SilentlyContinue
            throw 'Контрольная сумма SHA256 не совпала'
        }
        if (Test-Path -LiteralPath $dest -PathType Leaf) { Remove-Item -LiteralPath $dest -Force }
        Move-Item -LiteralPath $tmp -Destination $dest -Force
        Invoke-Api -Method 'Post' -Path ("/transfer/$tid/done") -BodyObj @{ sha256 = $hash } -Retries 6 -TimeoutSec 60 | Out-Null
        Write-Log ("Получен файл: " + $payload.path + " (" + $total + " байт)")
    } catch {
        if ($fs) { $fs.Dispose() }
        Remove-Item -LiteralPath $tmp -Force -ErrorAction SilentlyContinue
        $msg = $_.Exception.Message
        Fail-Transfer -TransferId $tid -Message $msg
        Write-Log ("Ошибка приёма файла " + $payload.path + ": " + $msg)
    }
    return $null
}

function Invoke-BridgeCommand {
    param($cmd)
    $payload = $cmd.payload
    if ($null -eq $payload) { $payload = @{} }
    switch ([string]$cmd.type) {
        'ping'          { return @{ ok = $true; info = $Info; time = (Get-Date).ToString('s') } }
        'stat'          { return Invoke-Stat $payload }
        'list'          { return Get-Listing $payload }
        'mkdir'         { return New-Dir $payload }
        'delete'        { return Remove-BridgeItem $payload }
        'move'          { return Move-BridgeItem $payload }
        'download_init' { return Invoke-DownloadTransfer -cmd $cmd -payload $payload }
        'upload_init'   { return Invoke-UploadTransfer -cmd $cmd -payload $payload }
        'exec'          { return Start-ExecTask -cmd $cmd -payload $payload }
        'exec_start'    { return Start-BackgroundProc -payload $payload }
        'proc_read'     { return Read-BackgroundProc -payload $payload }
        'proc_write'    { return Write-BackgroundProc -payload $payload }
        'proc_kill'     { return Kill-BackgroundProc -payload $payload }
        'proc_list'     { return List-BackgroundProcs }
        'read_text'     { return Read-TextFile -payload $payload }
        'write_text'    { return Write-TextFileCmd -payload $payload }
        'restart'       { return Invoke-Restart -cmd $cmd }
        default         { throw ("Неизвестная команда: " + $cmd.type) }
    }
}

import { db } from '@/lib/db'
import {
  requireBridgeAuth,
  jerr,
  jok,
  jappErr,
  sweep,
  sleep,
  clampPath,
  ACTION_WAIT_MS,
} from '@/lib/bridge'

export const dynamic = 'force-dynamic'

const OPS: Record<string, string> = {
  ping: 'ping',
  stat: 'stat',
  list: 'list',
  mkdir: 'mkdir',
  delete: 'delete',
  move: 'move',
}

/**
 * POST /api/bridge/action
 * Body: { op: 'ping'|'stat'|'list'|'mkdir'|'delete'|'move', path?, to?, recursive? }
 * Ставит команду в очередь и ждёт выполнения агентом (до ACTION_WAIT_MS).
 */
export async function POST(request: Request) {
  const token = await requireBridgeAuth(request)
  if (!token) return jerr('Неверный токен моста', 401)

  const body = (await request.json().catch(() => ({}))) as {
    op?: string
    path?: string
    to?: string
    recursive?: boolean
  }
  const op = String(body.op ?? '')
  const type = OPS[op]
  if (!type) return jerr(`Неизвестная операция: ${op || '(пусто)'}`)

  let payload: Record<string, unknown>
  switch (type) {
    case 'ping':
      payload = {}
      break
    case 'list':
      payload = { path: clampPath(body.path), recursive: Boolean(body.recursive) }
      break
    case 'stat':
    case 'mkdir':
    case 'delete':
      payload = { path: clampPath(body.path) }
      if (!String(payload.path).trim() && type !== 'stat') {
        return jerr('Не указан путь')
      }
      break
    case 'move': {
      const from = clampPath(body.path)
      const to = clampPath(body.to)
      if (!from.trim() || !to.trim()) return jerr('Для перемещения нужны path и to')
      payload = { from, to }
      break
    }
    default:
      return jerr('Неизвестная операция')
  }

  const cmd = await db.bridgeCommand.create({
    data: { type, payload: JSON.stringify(payload) },
  })

  const deadline = Date.now() + ACTION_WAIT_MS
  let ticks = 0
  while (Date.now() < deadline) {
    await sleep(350)
    ticks++
    if (ticks % 12 === 0) await sweep()
    const c = await db.bridgeCommand.findUnique({ where: { id: cmd.id } })
    if (!c) break
    if (c.status === 'done') {
      return jok({
        data: c.result ? (JSON.parse(c.result) as unknown) : null,
        commandId: c.id,
      })
    }
    if (c.status === 'failed') {
      return jappErr(c.error || 'Ошибка выполнения на ПК')
    }
  }
  return jerr('Таймаут ожидания ответа моста (мост офлайн или занят)', 504)
}

/**
 * POST /api/bridge/agent/legacy-restart — перезапуск агента v1.x через watchdog.
 *
 * Как работает: агент v1 отвечает на poll каждые ~1с и обновляет heartbeat.
 * Если поставить флаг hangChunksUntil и инициировать вытягивание маленького
 * файла, агент войдёт в chunk-цикл, а сервер не будет отвечать на чанк.
 * Через 6 повторов по 120с (~12 мин без heartbeat) внутренний watchdog агента
 * убьёт процесс, а zai-bridge-setup.bat перезапустит его за 3 секунды —
 * уже с обновлённым bridge.ps1.
 *
 * Body: { hangSec?: number (default 900), probePath?: string (файл для зависания) }
 *       { clear: true } — снять флаг без зависания.
 */
export async function POST(request: Request) {
  const token = await requireBridgeAuth(request)
  if (!token) return jerr('Неверный токен моста', 401)

  const body = (await request.json().catch(() => ({}))) as {
    hangSec?: unknown
    probePath?: unknown
    clear?: boolean
  }

  if (body.clear === true) {
    await setSetting('hangChunksUntil', '')
    return jok({ cleared: true })
  }

  // Уже висит?
  const existing = await getSetting('hangChunksUntil')
  if (existing && Date.now() < new Date(existing).getTime()) {
    return jok({ alreadyHanging: true, until: existing })
  }

  let hangSec = Number(body.hangSec)
  if (!Number.isFinite(hangSec) || hangSec <= 0) hangSec = 900
  hangSec = Math.min(hangSec, 1800)

  const probePath = clampPath(body.probePath) || 'bridge_probe.txt'

  await setSetting('hangChunksUntil', new Date(Date.now() + hangSec * 1000).toISOString())

  // Инициируем pc2sb-передачу маленького файла, чтобы агент зашёл в chunk-цикл
  const stagedName = makeStagedName(probePath)
  const transfer = await db.bridgeTransfer.create({
    data: { direction: 'pc2sb', path: probePath, stagedName, status: 'pending' },
  })
  await db.bridgeCommand.create({
    data: {
      type: 'download_init',
      payload: JSON.stringify({ transferId: transfer.id, path: probePath }),
      transferId: transfer.id,
    },
  })

  return jok({
    hangSec,
    until: new Date(Date.now() + hangSec * 1000).toISOString(),
    probePath,
    transferId: transfer.id,
    note: 'Агент v1 зависнет на чанке; watchdog убьёт процесс ~через 10-12 мин, bat-лаунчер перезапустит с новым bridge.ps1',
  })
}

  clampCmd,
  clampPath,

  EXEC_MAX_TIMEOUT_SEC,
} from '@/lib/bridge'

/**
 * POST /api/bridge/exec — выполнить команду на ПК (синхронно, с ожиданием).
 * Body: { cmd, cwd?, timeoutSec? (5..600, default 120), env?: {K:V}, waitMs? (сколько ждать ответа роуту) }
 * Если за waitMs ответ не пришёл — { ok: true, pending: true, commandId } (допросить через GET /api/bridge/command/<id>).
 */
export async function POST(request: Request) {
  const token = await requireBridgeAuth(request)
  if (!token) return jerr('Неверный токен моста', 401)

  const body = (await request.json().catch(() => ({}))) as {
    cmd?: unknown
    cwd?: unknown
    timeoutSec?: unknown
    env?: unknown
    waitMs?: unknown
  }

  const cmd = clampCmd(body.cmd)
  if (!cmd.trim()) return jerr('Пустая команда')

  let timeoutSec = Number(body.timeoutSec)
  if (!Number.isFinite(timeoutSec) || timeoutSec <= 0) timeoutSec = 120
  if (timeoutSec > EXEC_MAX_TIMEOUT_SEC) timeoutSec = EXEC_MAX_TIMEOUT_SEC
  if (timeoutSec < 5) timeoutSec = 5

  const env = sanitizeEnv(body.env)
  if (env === null) return jerr('Некорректный env (до 32 пар, ключи до 256, значения до 1024 символов)')

  const cwd = clampPath(body.cwd)
  const payload = { cmd, cwd, timeoutSec, env }

  const command = await db.bridgeCommand.create({
    data: { type: 'exec', payload: JSON.stringify(payload) },
  })

  // Сколько ждём в этом HTTP-запросе: по умолчанию timeoutSec + 30с, но не более 5 минут
  let waitMs = Number(body.waitMs)
  if (!Number.isFinite(waitMs) || waitMs <= 0) waitMs = Math.min(timeoutSec * 1000 + 30_000, 300_000)
  waitMs = Math.max(5_000, Math.min(waitMs, 300_000))

  const deadline = Date.now() + waitMs
  let ticks = 0
  while (Date.now() < deadline) {
    await sleep(350)
    ticks++
    if (ticks % 20 === 0) await sweep()
    const c = await db.bridgeCommand.findUnique({ where: { id: command.id } })
    if (!c) break
    if (c.status === 'done') {
      return jok({
        data: c.result ? (JSON.parse(c.result) as unknown) : null,
        commandId: c.id,
        pending: false,
      })
    }
    if (c.status === 'failed') {
      return jappErr(c.error || 'Ошибка выполнения на ПК')
    }
  }
  // Время вышло — команда ещё исполняется, отдаём pending
  return jok({ pending: true, commandId: command.id, timeoutSec })
}

  const size = fs.statSync(p).size

  try {
    const stream = toWebStream(createReadStream(p))
    return new Response(stream, {
      headers: {
        'content-type': 'application/octet-stream',
        'content-length': String(size),
        'content-disposition': `attachment; filename="${name}"`,
        'cache-control': 'no-store',
      },
    })
  } catch {
    const buf = await fs.promises.readFile(p)
    return new Response(new Uint8Array(buf), {
      headers: {
        'content-type': 'application/octet-stream',
        'content-length': String(buf.length),
        'content-disposition': `attachment; filename="${name}"`,
        'cache-control': 'no-store',
      },
    })
  }
}

/**
 * PUT /api/bridge/file/[name] — положить файл в песочницу (raw-байты в теле).
 */
export async function PUT(
  request: Request,
  { params }: { params: Promise<{ name: string }> }
) {
  const token = await requireBridgeAuth(request)
  if (!token) return jerr('Неверный токен моста', 401)

  const { name: rawName } = await params
  const name = sanitizeStagedName(decodeURIComponent(rawName))
  if (!name) return jerr('Некорректное имя файла (допустимы A-Z a-z 0-9 . _ -, без точек в начале)')

  ensureDirs()
  const p = stagedPath(name)

  try {
    if (request.body) {
      const nodeStream = Readable.fromWeb(request.body as unknown as NodeWebReadableStream<Uint8Array>)
      await pipeline(nodeStream, createWriteStream(p))
    } else {
      await fs.promises.writeFile(p, Buffer.alloc(0))
    }
  } catch (e) {
    return jerr(`Не удалось сохранить файл: ${e instanceof Error ? e.message : String(e)}`, 500)
  }

  const size = fs.statSync(p).size
  if (size > MAX_FILE_SIZE) {
    await fs.promises.rm(p, { force: true })
    return jerr('Файл больше 2 ГБ — не поддерживается')
  }
  const sha256 = await sha256File(p)

  return jok({ name, size, sha256 })
}

async function walk(dir: string, base: string, out: { path: string; content: string; size: number }[]) {
  if (out.length >= MAX_FILES) return
  let entries: fs.Dirent[]
  try {
    entries = await fs.promises.readdir(dir, { withFileTypes: true })
  } catch {
    return
  }
  // Сначала файлы, потом подпапки; сортировка по имени
  entries.sort((a, b) => a.name.localeCompare(b.name))
  for (const entry of entries) {
    if (out.length >= MAX_FILES) return
    const full = path.join(dir, entry.name)
    const rel = path.relative(base, full).split(path.sep).join('/')
    if (entry.isDirectory()) {
      await walk(full, base, out)
    } else if (entry.isFile() && /\.md$/i.test(entry.name)) {
      try {
        const stat = await fs.promises.stat(full)
        if (stat.size > MAX_FILE_BYTES) {
          out.push({ path: rel, content: `-- файл слишком большой (${stat.size} байт) --`, size: stat.size })
          continue
        }
        const content = await fs.promises.readFile(full, 'utf8')
        out.push({ path: rel, content, size: stat.size })
      } catch {
        /* пропускаем нечитаемое */
      }
    }
  }
}

      },

    )
  }

  await sweep()

    where: { status: 'pending' },
    orderBy: { createdAt: 'asc' },
    take: 8,

        })),

import { db } from '@/lib/db'
import {
  requireBridgeAuth,
  jerr,
  jok,
  jappErr,
  sweep,
  sleep,
  clampCmd,
  clampPath,
  sanitizeEnv,
  ACTION_WAIT_MS,
  PROC_MAX_OUTPUT_CHARS,
} from '@/lib/bridge'

export const dynamic = 'force-dynamic'

interface ProcPayload {
  cmd?: string
  cwd?: string
  env?: Record<string, string>
  handle?: string
  fromOut?: number
  fromErr?: number
  maxChars?: number
  input?: string
  newline?: boolean
}

async function queueAndWait(type: string, payload: ProcPayload) {
  const cmd = await db.bridgeCommand.create({
    data: { type, payload: JSON.stringify(payload) },
  })
  const deadline = Date.now() + ACTION_WAIT_MS
  let ticks = 0
  while (Date.now() < deadline) {
    await sleep(300)
    ticks++
    if (ticks % 15 === 0) await sweep()
    const c = await db.bridgeCommand.findUnique({ where: { id: cmd.id } })
    if (!c) break
    if (c.status === 'done') {
      return jok({
        data: c.result ? (JSON.parse(c.result) as unknown) : null,
        commandId: c.id,
      })
    }
    if (c.status === 'failed') {
      return jappErr(c.error || 'Ошибка выполнения на ПК')
    }
  }
  return jerr('Таймаут ожидания ответа моста (мост офлайн или занят)', 504)
}

/**
 * POST /api/bridge/proc — управление фоновыми процессами на ПК.
 *
 * op=start: { cmd, cwd?, env? }                          -> { handle, pid, cmd }
 * op=read:  { handle, fromOut?, fromErr?, maxChars? }     -> { stdout, stderr, nextOut, nextErr, running, exitCode?, ... }
 * op=write: { handle, input, newline? }                   -> { ok } (stdin)
 * op=kill:  { handle }                                    -> { ok }
 * op=list:  {}                                            -> { procs: [...] }
 */
export async function POST(request: Request) {
  const token = await requireBridgeAuth(request)
  if (!token) return jerr('Неверный токен моста', 401)

  const body = (await request.json().catch(() => ({}))) as ProcPayload & { op?: string }
  const op = String(body.op ?? '')

  switch (op) {
    case 'start': {
      const cmd = clampCmd(body.cmd)
      if (!cmd.trim()) return jerr('Пустая команда')
      const env = sanitizeEnv(body.env)
      if (env === null) return jerr('Некорректный env')
      const payload: ProcPayload = { cmd, cwd: clampPath(body.cwd), env }
      return queueAndWait('exec_start', payload)
    }
    case 'read': {
      const handle = String(body.handle ?? '').trim()
      if (!handle) return jerr('Не указан handle процесса')
      const fromOut = Number.isInteger(Number(body.fromOut)) && Number(body.fromOut) >= 0 ? Number(body.fromOut) : 0
      const fromErr = Number.isInteger(Number(body.fromErr)) && Number(body.fromErr) >= 0 ? Number(body.fromErr) : 0
      let maxChars = Number(body.maxChars)
      if (!Number.isFinite(maxChars) || maxChars <= 0) maxChars = 100_000
      maxChars = Math.min(maxChars, PROC_MAX_OUTPUT_CHARS)
      return queueAndWait('proc_read', { handle, fromOut, fromErr, maxChars })
    }
    case 'write': {
      const handle = String(body.handle ?? '').trim()
      if (!handle) return jerr('Не указан handle процесса')
      const input = String(body.input ?? '').slice(0, 100_000)
      const payload: ProcPayload = { handle, input, newline: body.newline !== false }
      return queueAndWait('proc_write', payload)
    }
    case 'kill': {
      const handle = String(body.handle ?? '').trim()
      if (!handle) return jerr('Не указан handle процесса')
      return queueAndWait('proc_kill', { handle })
    }
    case 'list': {
      return queueAndWait('proc_list', {})
    }
    default:
      return jerr(`Неизвестная операция: ${op || '(пусто)'}. Доступны: start, read, write, kill, list`)
  }
}

import {
  requireBridgeAuth,
  jerr,
  jok,
  jappErr,
  sweep,
  sleep,
  clampPath,
  ACTION_WAIT_MS,
  TEXT_MAX_CHARS,
} from '@/lib/bridge'

export const dynamic = 'force-dynamic'

interface TextPayload {
  path?: string
  maxChars?: number
  content?: string
  append?: boolean
}

async function queueAndWait(type: string, payload: TextPayload) {
  const cmd = await db.bridgeCommand.create({
    data: { type, payload: JSON.stringify(payload) },
  })
  const deadline = Date.now() + ACTION_WAIT_MS
  let ticks = 0
  while (Date.now() < deadline) {
    await sleep(300)
    ticks++
    if (ticks % 15 === 0) await sweep()
    const c = await db.bridgeCommand.findUnique({ where: { id: cmd.id } })
    if (!c) break
    if (c.status === 'done') {
      return jok({
        data: c.result ? (JSON.parse(c.result) as unknown) : null,
        commandId: c.id,
      })
    }
    if (c.status === 'failed') {
      return jappErr(c.error || 'Ошибка выполнения на ПК')
    }
  }
  return jerr('Таймаут ожидания ответа моста (мост офлайн или занят)', 504)
}

/**
 * POST /api/bridge/text — быстрые текстовые операции с файлами на ПК.
 *
 * op=read:  { path, maxChars? }          -> { content, truncated, sizeBytes, encoding }
 * op=write: { path, content, append? }   -> { ok, bytes, created }
 *
 * Файлы пишутся в UTF-8 без BOM. Только внутри папки моста.
 */
export async function POST(request: Request) {
  const token = await requireBridgeAuth(request)
  if (!token) return jerr('Неверный токен моста', 401)

  const body = (await request.json().catch(() => ({}))) as TextPayload & { op?: string }
  const op = String(body.op ?? '')

  switch (op) {
    case 'read': {
      const path = clampPath(body.path)
      if (!path.trim()) return jerr('Не указан путь файла')
      let maxChars = Number(body.maxChars)
      if (!Number.isFinite(maxChars) || maxChars <= 0) maxChars = 200_000
      maxChars = Math.min(maxChars, TEXT_MAX_CHARS)
      return queueAndWait('read_text', { path, maxChars })
    }
    case 'write': {
      const path = clampPath(body.path)
      if (!path.trim()) return jerr('Не указан путь файла')
      const content = String(body.content ?? '').slice(0, TEXT_MAX_CHARS)
      return queueAndWait('write_text', { path, content, append: body.append === true })
    }
    default:
      return jerr(`Неизвестная операция: ${op || '(пусто)'}. Доступны: read, write`)
  }
}

async function loadTransfer(id: string) {

/**
 * Сервисный режим: «подвесить» ответ на чанк, чтобы агент v1.x остановился
 * в chunk-цикле и его внутренний watchdog перезапустил процесс (подхватится
 * уже обновлённый bridge.ps1). Флаг hangChunksUntil ставится через
 * POST /api/bridge/agent/legacy-restart и снимается автоматически по протухании.
 */
async function hangIfRequested(): Promise<void> {
  const untilRaw = await getSetting('hangChunksUntil')
  if (!untilRaw) return
  const until = new Date(untilRaw).getTime()
  if (!Number.isFinite(until)) {
    await setSetting('hangChunksUntil', '')
    return
  }
  while (Date.now() < until) {
    await sleep(5000)
  }
  await setSetting('hangChunksUntil', '')
}

/**
 * POST /api/bridge/transfer/[id]/chunk — агент на ПК отправляет чанк файла (pc2sb).
 * Два формата (v2.3):
 *  1) бинарный: Content-Type: application/octet-stream, ?index=N[&gz=1],
 *     тело — сырые (или gzip) байты чанка;
 *  2) легаси JSON: { index: number, dataB64: string } (v2.2 и старше).
 */
export async function POST(
  request: Request,
  { params }: { params: Promise<{ id: string }> }
) {
  const token = await requireBridgeAuth(request)
  if (!token) return jerr('Неверный токен моста', 401)

  const { id } = await params
  const t = await loadTransfer(id)
  if (!t) return jerr('Передача не найдена', 404)
  if (t.direction !== 'pc2sb') return jerr('Неверное направление передачи', 400)
  if (t.status !== 'active' && t.status !== 'pending') {
    return jerr(`Передача уже завершена (${t.status})`, 400)
  }

  await hangIfRequested()

  const url = new URL(request.url)
  const contentType = request.headers.get('content-type') || ''
  let index: number
  let buf: Buffer

  if (contentType.includes('application/octet-stream')) {
    // --- v2.3: бинарный чанк ---
    index = Number(url.searchParams.get('index'))
    if (!Number.isInteger(index) || index < 0) return jerr('Некорректный index')
    const raw = Buffer.from(await request.arrayBuffer())
    if (raw.length === 0) return jerr('Пустой чанк')
    if (raw.length > 16 * 1024 * 1024) return jerr('Чанк слишком большой', 413)
    if (url.searchParams.get('gz') === '1') {
      try {
        buf = zlib.gunzipSync(raw)
      } catch {
        return jerr('Некорректный gzip-чанк')
      }
      if (buf.length > 16 * 1024 * 1024) return jerr('Чанк слишком большой', 413)
    } else {
      buf = raw
    }
  } else {
    // --- легаси JSON (v2.2-) ---
    const body = (await request.json().catch(() => ({}))) as {
      index?: number
      dataB64?: string
    }
    index = Number(body.index)
    if (!Number.isInteger(index) || index < 0) return jerr('Некорректный index')
    const dataB64 = String(body.dataB64 ?? '')
    if (!dataB64) return jerr('Пустой чанк')
    if (dataB64.length > MAX_CHUNK_B64) return jerr('Чанк слишком большой', 413)
    try {
      buf = Buffer.from(dataB64, 'base64')
    } catch {
      return jerr('Некорректный base64')
    }
    if (buf.length === 0) return jerr('Пустой чанк')
    if (buf.length > 16 * 1024 * 1024) return jerr('Чанк слишком большой', 413)
  }

  if (index !== t.chunksDone) {
    return jerr(`Нарушен порядок чанков: ожидался ${t.chunksDone}, получен ${index}`)
  }

  ensureDirs()
  await appendChunk(id, buf)

  const bytesDone = t.bytesDone + buf.length
  if (t.size > 0 && bytesDone > t.size) {
    await db.bridgeTransfer.update({
      where: { id },
      data: { status: 'failed', error: 'Получено больше байт, чем заявлено' },
    })
    await fs.promises.rm(tmpPath(id), { force: true }).catch(() => {})
    return jerr('Превышен заявленный размер файла')
  }

  await db.bridgeTransfer.update({
    where: { id },
    data: { bytesDone, chunksDone: index + 1, status: 'active' },
  })
  await setSetting('lastSeenAt', new Date().toISOString())

  return jok({ received: bytesDone })
}

/**
 * GET /api/bridge/transfer/[id]/chunk?index=N — агент на ПК ТЯНЕТ чанк (sb2pc).
 * Два формата (v2.3):
 *  1) бинарный: &bin=1[&gz=1] — сырые байты (или gzip, если выгодно),
 *     метаданные в заголовках x-index / x-last / x-eof / x-gz;
 *  2) легаси JSON: { index, dataB64, last } / { eof: true } (v2.2 и старше).
 */
export async function GET(
  request: Request,
  { params }: { params: Promise<{ id: string }> }
) {
  const token = await requireBridgeAuth(request)
  if (!token) return jerr('Неверный токен моста', 401)

  const { id } = await params
  const t = await loadTransfer(id)
  if (!t) return jerr('Передача не найдена', 404)
  if (t.direction !== 'sb2pc') return jerr('Неверное направление передачи', 400)
  if (t.status !== 'active' && t.status !== 'pending') {
    return jok({ eof: true, status: t.status })
  }

  const src = stagedPath(t.stagedName)
  if (!fs.existsSync(src)) {
    await db.bridgeTransfer.update({
      where: { id },
      data: { status: 'failed', error: 'Файл-источник отсутствует в песочнице' },
    })
    return jerr('Файл-источник отсутствует в песочнице', 410)
  }
  const size = fs.statSync(src).size

  const url = new URL(request.url)
  const bin = url.searchParams.get('bin') === '1'

  if (size === 0) {
    if (bin) return binaryChunkResponse(Buffer.alloc(0), 0, true, false)
    return jok({ eof: true })
  }

  const index = Number(url.searchParams.get('index'))
  if (!Number.isInteger(index) || index < 0) return jerr('Некорректный index')
  const totalChunks = Math.ceil(size / CHUNK_SIZE)
  const isLast = index === totalChunks - 1
  if (index >= totalChunks) {
    if (bin) return binaryChunkResponse(Buffer.alloc(0), index, true, false)
    return jok({ eof: true })
  }

  const start = index * CHUNK_SIZE
  const len = Math.min(CHUNK_SIZE, size - start)
  const fh = await fs.promises.open(src, 'r')
  try {
    const buf = Buffer.alloc(len)
    await fh.read(buf, 0, len, start)
    await setSetting('lastSeenAt', new Date().toISOString())
    if (bin) {
      let body = buf
      let gz = false
      if (url.searchParams.get('gz') === '1' && buf.length > 512) {
        const packed = zlib.gzipSync(buf, { level: 6 })
        if (packed.length < buf.length - 64) {
          body = packed
          gz = true
        }
      }
      return binaryChunkResponse(body, index, false, gz, isLast)
    }
    return jok({
      index,
      dataB64: buf.toString('base64'),
      last: isLast,
    })
  } finally {
    await fh.close()
  }
}

function binaryChunkResponse(body: Buffer, index: number, eof: boolean, gz: boolean, last = false) {
  return new Response(new Uint8Array(body), {
    headers: {
      'content-type': 'application/octet-stream',
      'content-length': String(body.length),
      'x-index': String(index),
      'x-eof': eof ? '1' : '0',
      'x-gz': gz ? '1' : '0',
      'x-last': last ? '1' : '0',
      'cache-control': 'no-store',
    },
  })
}

/**
 * POST /api/bridge/transfer/[id]/complete — агент на ПК завершил отправку файла (pc2sb).
 * Body: { sha256: string }
 * Сервер проверяет размер и хэш, перемещает tmp -> db/bridge-files/<stagedName>.
 */
export async function POST(
  request: Request,
  { params }: { params: Promise<{ id: string }> }
) {
  const token = await requireBridgeAuth(request)
  if (!token) return jerr('Неверный токен моста', 401)

  const { id } = await params
  const t = await db.bridgeTransfer.findUnique({ where: { id } })
  if (!t) return jerr('Передача не найдена', 404)
  if (t.direction !== 'pc2sb') return jerr('Неверное направление передачи', 400)
  if (t.status === 'complete') return jok({ stagedName: t.stagedName, sha256: t.sha256, size: t.size })
  if (t.status !== 'active' && t.status !== 'pending') {
    return jerr(`Передача уже завершена (${t.status})`, 400)
  }

  const body = (await request.json().catch(() => ({}))) as { sha256?: string }
  const claimed = String(body.sha256 ?? '').toLowerCase()

  ensureDirs()
  const tp = tmpPath(id)

  const fail = async (error: string) => {
    await db.bridgeTransfer.update({ where: { id }, data: { status: 'failed', error } })
    await fs.promises.rm(tp, { force: true }).catch(() => {})
    return jerr(error)
  }

  // Пустой файл — чанков не было
  if (!fs.existsSync(tp)) {
    if (t.size === 0) {
      await fs.promises.writeFile(stagedPath(t.stagedName), Buffer.alloc(0))
      await db.bridgeTransfer.update({
        where: { id },
        data: {
          status: 'complete',
          bytesDone: 0,
          size: 0,
          sha256: claimed || null,
          error: null,
        },
      })
      await setSetting('lastSeenAt', new Date().toISOString())
      return jok({ stagedName: t.stagedName, size: 0 })
    }
    return fail('Временный файл не найден (передача прервана)')
  }

  const actualSize = fs.statSync(tp).size
  if (t.size > 0 && actualSize !== t.size) {
    return fail(`Размер не совпал: заявлено ${t.size}, получено ${actualSize}`)
  }

  const hash = await sha256File(tp)
  if (claimed && claimed !== hash) {
    return fail('Контрольная сумма SHA256 не совпала — файл повреждён при передаче')
  }

  await fs.promises.rm(stagedPath(t.stagedName), { force: true })
  await fs.promises.rename(tp, stagedPath(t.stagedName))
  await db.bridgeTransfer.update({
    where: { id },
    data: {
      status: 'complete',
      bytesDone: actualSize,
      size: t.size > 0 ? t.size : actualSize,
      sha256: hash,
      error: null,
    },
  })
  await setSetting('lastSeenAt', new Date().toISOString())

  return jok({ stagedName: t.stagedName, sha256: hash, size: actualSize })
}

const BASE = "/api/bridge";

interface ApiEnvelope {
  ok: boolean;
  error?: string;
}

interface RequestOptions {
  method?: "GET" | "POST" | "PUT" | "DELETE";
  body?: string;
  token?: string;
}

async function request<T extends ApiEnvelope>(path: string, options: RequestOptions = {}): Promise<T> {
  const headers = new Headers();
  if (options.token) headers.set("x-bridge-token", options.token);
  if (options.body !== undefined) headers.set("content-type", "application/json");
  try {
    const res = await fetch(path, { method: options.method ?? "GET", headers, body: options.body, cache: "no-store" });
    const text = await res.text();
    let data: unknown = null;
    try {
      data = text ? JSON.parse(text) : null;
    } catch {
      data = null;
    }
    if (data !== null && typeof data === "object" && "ok" in (data as Record<string, unknown>)) {
      const parsed = data as T;
      if (parsed.ok === false && !parsed.error) {
        parsed.error = `Сервер вернул ошибку (${res.status})`;
      }
      return parsed;
    }
    return { ok: false, error: `Неожиданный ответ сервера (${res.status})` } as T;
  } catch {
    return { ok: false, error: "Не удалось связаться с сервером" } as T;
  }
}

export async function uploadStagedFile(
  token: string,
  name: string,
  file: Blob,
): Promise<{ ok: boolean; error?: string }> {
  try {
    const res = await fetch(`${BASE}/file/${encodeURIComponent(name)}`, {
      method: "PUT",
      headers: { "x-bridge-token": token },
      body: file,

    try {
      const data = JSON.parse(text) as { ok?: boolean; error?: string };
      if (data.ok === true) return { ok: true };

    } catch {

    }

export type BlobResult = { ok: true; blob: Blob; filename: string | null } | { ok: false; error: string };

/** Загрузка бинарника с заголовком токена (файл настройки, staging-файл). */
export async function downloadWithToken(url: string, token: string): Promise<BlobResult> {
  try {
    const res = await fetch(url, { headers: { "x-bridge-token": token }, cache: "no-store" });
    if (!res.ok) {
      let error = `Ошибка загрузки (${res.status})`;
      try {
        const data = (await res.json()) as { error?: string };
        if (data?.error) error = data.error;
      } catch {
        // тело не JSON — оставляем статус
      }
      return { ok: false, error };
    }
    const disposition = res.headers.get("content-disposition") ?? "";
    const match = /filename\*?=(?:UTF-8'')?"?([^";]+)"?/i.exec(disposition);
    const filename = match ? decodeURIComponent(match[1]) : null;
    return { ok: true, blob: await res.blob(), filename };
  } catch {
    return { ok: false, error: "Не удалось связаться с сервером" };
  }
}

/** Ожидание завершения async-передачи (poll /transfers). */
export async function waitForTransfer(
  token: string,
  transferId: string,
  timeoutMs = 240_000,
  intervalMs = 2_000,
): Promise<{ status: "complete" | "failed" | "timeout"; error?: string | null }> {
  const deadline = Date.now() + timeoutMs;
  while (Date.now() < deadline) {
    const res = await fetchTransfers(token);
    if (res.ok && res.transfers) {
      const transfer = res.transfers.find((item) => item.id === transferId);
      if (transfer) {
        if (transfer.status === "complete") return { status: "complete" };
        if (transfer.status === "failed") return { status: "failed", error: transfer.error };
      }
    }
    await new Promise((resolve) => setTimeout(resolve, intervalMs));
  }
  return { status: "timeout" };
}

      for (const file of files) {
        const stagedName = sanitizeFileName(file.name);
        const targetPath = joinPath(pathRef.current, stagedName);
        const upload = await uploadStagedFile(token, stagedName, file);
        if (!upload.ok) {
          failures.push(`${file.name}: ${upload.error ?? "ошибка загрузки"}`);
          continue;
        }
        const push = await pushToPc(token, stagedName, targetPath);
        if (push.ok) {
          sent += 1;
        } else {
          failures.push(`${file.name}: ${push.error ?? "мост не принял файл"}`);
        }
      }
    } finally {
      setUploading(false);
    }
    if (sent > 0) {
      toast({
        title: sent === 1 ? "Файл отправляется на ПК" : `Отправляется файлов: ${sent}`,
        description: "Прогресс — в карточке «Передачи»",
      });
    }
    if (failures.length > 0) {
      toast({
        title: "Не удалось отправить некоторые файлы",
        description: failures.join(" • ").slice(0, 300),
        variant: "destructive",
      });
    }
    handleRefresh();
  };

  /* ------------------------------- Скачивание ------------------------------ */

  const handleDownload = async (entry: FileEntry) => {
    if (!token || downloading) return;
    setDownloading(entry.path);
    try {
      const pull = await pullFromPc(token, entry.path);
      if (!pull.ok || !pull.transferId || !pull.stagedName) {
        toast({
          title: "Не удалось начать загрузку",
          description: pull.error ?? "Мост не ответил",
          variant: "destructive",
        });
        return;
      }
      const result = await waitForTransfer(token, pull.transferId);
      if (result.status === "complete") {
        const fileRes = await downloadWithToken(
          `/api/bridge/file/${encodeURIComponent(pull.stagedName)}`,
          token,
        );
        if (fileRes.ok) {
          saveBlob(fileRes.blob, entry.name);
          toast({ title: "Файл загружен", description: entry.name });
        } else {
          toast({ title: "Не удалось скачать файл", description: fileRes.error, variant: "destructive" });
        }
      } else if (result.status === "failed") {
        toast({
          title: "Передача не удалась",
          description: result.error ?? "Мост вернул ошибку",
          variant: "destructive",
        });
      } else {
        toast({
          title: "Передача ещё идёт",
          description: "Следите за прогрессом в карточке «Передачи»",
        });
      }
    } finally {
      setDownloading(null);
    }
  };

  /* --------------------------------- Удаление ------------------------------ */

  const handleDelete = async () => {
    if (!deleteTarget || !token || deleteBusy) return;
    setDeleteBusy(true);
    try {
      const res = await bridgeAction(token, {
        op: "delete",
        path: deleteTarget.path,
        recursive: deleteTarget.isDir,
      });
      if (res.ok) {
        toast({ title: "Удалено", description: deleteTarget.name });
        setDeleteTarget(null);
        handleRefresh();
      } else {
        toast({ title: "Не удалось удалить", description: res.error, variant: "destructive" });
      }
    } finally {
      setDeleteBusy(false);
    }
  };

    setMovePath(entry.path);

    }

    try {

      if (res.ok) {

  const segments = useMemo(() => path.split("/").filter(Boolean), [path]);

  return (

      </CardHeader>

      <CardContent className="flex flex-1 flex-col gap-4">
        {/* Навигация */}

          {path !== "" && (

            </Button>

        open={deleteTarget !== null}
        onOpenChange={(open) => {
          if (!open) setDeleteTarget(null);
        }}
      >
        <AlertDialogContent className={`${DIALOG_DARK}`}>
          <AlertDialogHeader>
            <AlertDialogTitle>Удалить «{deleteTarget?.name}»?</AlertDialogTitle>
            <AlertDialogDescription className="text-zinc-400">
              {deleteTarget?.isDir

                : "Файл будет удалён с вашего ПК."}{" "}

          <AlertDialogFooter>
            <AlertDialogCancel className="border-zinc-700 bg-transparent text-zinc-200 hover:bg-zinc-800 hover:text-zinc-100">
              Отмена
            </AlertDialogCancel>
            <AlertDialogAction
              className="bg-red-600 text-white hover:bg-red-500"
              onClick={(event) => {
                event.preventDefault();
                void handleDelete();
              }}
            >
              {deleteBusy && <Loader2 className="animate-spin" aria-hidden="true" />}
              Удалить
            </AlertDialogAction>
          </AlertDialogFooter>
        </AlertDialogContent>
      </AlertDialog>
    </Card>
  );
}

function appendExecResult(blocks: ConsoleBlock[], res: ExecResult): ConsoleBlock[] {
  const out = [...blocks];
  if (res.stdout) out.push({ id: 0, kind: "out", text: res.stdout.replace(/\r\n$/, "") });
  if (res.stderr) out.push({ id: 0, kind: "err", text: res.stderr.replace(/\r\n$/, "") });
  const parts = [`код ${res.code}`, formatDuration(res.durationMs)];
  if (res.timedOut) parts.push("ПРЕВЫШЕН ТАЙМАУТ");
  if (res.stdoutEnc) parts.push(res.stdoutEnc);
  if (res.stdoutTrunc || res.stderrTrunc) parts.push("вывод обрезан");
  out.push({ id: 0, kind: "meta", text: `── ${parts.join(" · ")} ──` });
  return out.slice(-MAX_BLOCKS);
}

        cursorsRef.current[res.data.handle] = { out: 0, err: 0 };
        pushBlocks([
          { kind: "meta", text: `Запущено в фоне: ${res.data.handle} (pid ${res.data.pid})` },
        ]);
        void refreshProcs();
      } else {
        pushBlocks([{ kind: "err", text: res.error ?? "Не удалось запустить" }]);
      }
      return;
    }

    setRunning(true);
    try {
      const res = await runExec(token, { cmd: text, timeoutSec: Number(timeoutSec) });
      if (res.ok && res.pending && res.commandId) {
        pushBlocks([
          { kind: "meta", text: `Команда ещё исполняется (id ${res.commandId}) — следим за ней…` },
        ]);
        pollCommand(res.commandId);
        return;
      }
      if (res.ok && res.data) {
        handleExecResult(res.data);
      } else {
        pushBlocks([{ kind: "err", text: res.error ?? "Ошибка выполнения" }]);
        setRunning(false);
      }
    } catch {
      pushBlocks([{ kind: "err", text: "Сетевая ошибка" }]);
      setRunning(false);
    }
  }, [
    cmd,
    running,
    token,
    online,
    bgMode,
    timeoutSec,
    pushBlocks,
    refreshProcs,
    pollCommand,
    handleExecResult,
  ]);

  const readProc = useCallback(
    async (handle: string) => {
      const cursor = cursorsRef.current[handle] ?? { out: 0, err: 0 };
      const res = await procRead(token, handle, cursor.out, cursor.err);
      if (!res.ok || !res.data) {
        pushBlocks([{ kind: "err", text: res.error ?? "Не удалось прочитать вывод" }]);
        return;
      }
      const d = res.data;
      cursorsRef.current[handle] = { out: d.nextOut, err: d.nextErr };
      const parts: { kind: BlockKind; text: string }[] = [];
      if (d.stdout) parts.push({ kind: "out", text: d.stdout.replace(/\r\n$/, "") });
      if (d.stderr) parts.push({ kind: "err", text: d.stderr.replace(/\r\n$/, "") });
      if (parts.length === 0) parts.push({ kind: "meta", text: `⟨${handle}⟩ новых строк нет` });
      if (!d.running) {
        parts.push({
          kind: "meta",
          text: `⟨${handle}⟩ процесс завершён — код ${d.exitCode ?? "?"} · ${formatDuration(d.durationMs ?? 0)}`,
        });
      }
      pushBlocks(parts);
    },
    [token, pushBlocks],
  );

  const killProc = useCallback(
    async (handle: string) => {
      const res = await procKill(token, handle);
      if (res.ok) {
        pushBlocks([{ kind: "meta", text: `⟨${handle}⟩ остановлен` }]);
        void refreshProcs();
      } else {
        toast({ title: "Не удалось остановить", description: res.error, variant: "destructive" });
      }
    },
    [token, pushBlocks, refreshProcs, toast],
  );

  const handleRestart = useCallback(async () => {
    setRestarting(true);
    try {
      const res = await restartAgent(token);
      if (res.ok) {
        toast({
          title: "Агент перезапускается",
          description: "Он вернётся онлайн через ~5 секунд",
        });
        pushBlocks([{ kind: "meta", text: "Перезапуск агента по команде из веб-интерфейса…" }]);
      } else {
        toast({ title: "Не удалось перезапустить", description: res.error, variant: "destructive" });
      }
    } finally {
      setRestarting(false);
      setRestartOpen(false);
    }
  }, [token, pushBlocks, toast]);

  return (
    <Card className={`${CARD_DARK} flex flex-col`}>
      <CardHeader>
        <div className="flex flex-wrap items-center justify-between gap-2">
          <div className="min-w-0">
            <CardTitle className="flex items-center gap-2 text-base">
              <Terminal className="size-4 text-emerald-500" aria-hidden="true" />

            <CardDescription className="text-zinc-500">

          </div>

              <Loader2 className="animate-spin" aria-hidden="true" />
            ) : (
              <RotateCw aria-hidden="true" />
            )}
            Перезапустить агент
          </Button>
        </div>
      </CardHeader>

      <CardContent className="flex flex-1 flex-col gap-4">
        {/* Ввод команды */}
        <div className="flex flex-col gap-2">
          <div className="flex flex-col gap-2 sm:flex-row">
            <div className="relative min-w-0 flex-1">
              <TextCursorInput
                className="pointer-events-none absolute left-3 top-1/2 size-4 -translate-y-1/2 text-zinc-600"
                aria-hidden="true"
              />
              <Input
                value={cmd}
                onChange={(e) => setCmd(e.target.value)}
                onKeyDown={(e) => {
                  if (e.key === "Enter") {
                    e.preventDefault();

                  }

                className={`${INPUT_DARK} pl-9 font-mono text-sm`}

              <Select value={timeoutSec} onValueChange={setTimeoutSec} disabled={!online || bgMode}>

function TransferProgress({ transfer }: { transfer: Transfer }) {
  const pct =
    transfer.size > 0 ? Math.min(100, Math.round((transfer.bytesDone / transfer.size) * 100)) : null;

  if (transfer.status === "failed") {
    return (
      <Progress
        value={pct ?? 100}
        className="h-1.5 bg-zinc-800 [&_[data-slot=progress-indicator]]:bg-red-500"
      />
    );
  }
  if (transfer.status === "complete") {
    return (
      <Progress
        value={100}
        className="h-1.5 bg-zinc-800 [&_[data-slot=progress-indicator]]:bg-emerald-500"
      />
    );
  }
  if (pct === null) {
    // size неизвестен — неопределённый прогресс
    return (
      <div
        className="h-1.5 w-full overflow-hidden rounded-full bg-zinc-800"
        role="progressbar"
        aria-label="Передача выполняется"
      >
        <div className="bridge-indeterminate h-full w-1/4 rounded-full bg-emerald-500" />
      </div>
    );
  }
  return (
    <Progress
      value={pct}
      className="h-1.5 bg-zinc-800 [&_[data-slot=progress-indicator]]:bg-emerald-500"
      aria-valuetext={`${pct}%`}
    />
  );
}

export function TransfersCard({ token }: TransfersCardProps) {

  const [hasActive, setHasActive] = useState(false);
  const hasActiveRef = useRef(false);

    if (!token) {
      return;
    }
    let cancelled = false;
    let timer: ReturnType<typeof setTimeout> | null = null;

    const tick = async () => {
      const res = await fetchTransfers(token);
      if (cancelled) return;
      if (res.ok && res.transfers) {
        setTransfers(res.transfers);
        const active = res.transfers.some((t) => t.status === "pending" || t.status === "active");
        hasActiveRef.current = active;
        setHasActive(active);
      }
      timer = setTimeout(() => void tick(), hasActiveRef.current ? 2_000 : 6_000);
    };

    void tick();
    return () => {
      cancelled = true;
      if (timer) clearTimeout(timer);
    };
  }, [token]);

  return (
    <Card className={`${CARD_DARK} flex h-full flex-col`}>

// Утилиты форматирования и работы с путями/файлами.

export function formatBytes(bytes: number): string {
  if (!Number.isFinite(bytes) || bytes < 0) return "—";
  if (bytes < 1024) return `${bytes} Б`;
  const units = ["КБ", "МБ", "ГБ", "ТБ"];
  let value = bytes / 1024;
  let unitIndex = 0;
  while (value >= 1024 && unitIndex < units.length - 1) {
    value /= 1024;
    unitIndex += 1;
  }
  return `${value.toLocaleString("ru-RU", { maximumFractionDigits: 1 })} ${units[unitIndex]}`;
}

/** HH:MM:SS из ISO-строки. */
export function formatClock(iso: string): string {
  const date = new Date(iso);
  if (Number.isNaN(date.getTime())) return "--:--:--";
  return date.toLocaleTimeString("ru-RU", { hour: "2-digit", minute: "2-digit", second: "2-digit" });
}

/** «01.01.2025 10:00» из mtime вида "2025-01-01 10:00" или ISO. */
export function formatMtime(mtime: string): string {
  if (!mtime) return "—";
  const normalized = mtime.includes("T") ? mtime : mtime.replace(" ", "T");
  const date = new Date(normalized);
  if (Number.isNaN(date.getTime())) return mtime;
  const day = date.toLocaleDateString("ru-RU", { day: "2-digit", month: "2-digit", year: "numeric" });
  const time = date.toLocaleTimeString("ru-RU", { hour: "2-digit", minute: "2-digit" });
  return `${day} ${time}`;
}

/** Сборка подпути от корня папки моста ("" = корень). */
export function joinPath(dir: string, name: string): string {
  if (!dir) return name;
  return `${dir}/${name}`;
}

/** Родительский подпуть: "a/b/c" -> "a/b", "a" -> "". */
export function parentPath(path: string): string {
  const index = path.lastIndexOf("/");
  return index === -1 ? "" : path.slice(0, index);
}

/** Санитизация имени для staging: только [A-Za-z0-9._-], до 80 символов. */
export function sanitizeFileName(name: string): string {
  const cleaned = name.replace(/[^A-Za-z0-9._-]+/g, "_").slice(0, 80);
  return cleaned.length > 0 ? cleaned : "file";
}

/** Сохранение Blob под указанным именем через объектную ссылку. */
export function saveBlob(blob: Blob, filename: string): void {
  const url = URL.createObjectURL(blob);
  const anchor = document.createElement("a");
  anchor.href = url;
  anchor.download = filename;
  anchor.rel = "noopener";
  document.body.appendChild(anchor);
  anchor.click();
  anchor.remove();
  window.setTimeout(() => URL.revokeObjectURL(url), 10_000);
}

/** Копирование в буфер обмена с запасным вариантом для старых браузеров. */
export async function copyToClipboard(text: string): Promise<boolean> {
  try {
    if (typeof navigator !== "undefined" && navigator.clipboard?.writeText) {
      await navigator.clipboard.writeText(text);
      return true;
    }
  } catch {
    // переходим к запасному варианту ниже
  }
  try {
    const area = document.createElement("textarea");
    area.value = text;
    area.setAttribute("readonly", "");
    area.style.position = "fixed";
    area.style.opacity = "0";
    document.body.appendChild(area);
    area.select();
    const copied = document.execCommand("copy");
    area.remove();
    return copied;
  } catch {
    return false;
  }
}

export function sanitizeStagedName(raw: unknown): string | null {
  const name = String(raw ?? '').trim()
  if (!name || name.length > 100) return null

/** Из человекочитаемого имени файла (возможно кириллица) делаем безопасное staged-имя */
export function makeStagedName(displayName: string): string {
  const base = path.basename(String(displayName || 'file'))
  const extMatch = path.extname(base).match(/^[A-Za-z0-9.]{0,12}/)
  const ext = extMatch ? extMatch[0] : ''
  let stem = base.slice(0, Math.max(0, base.length - ext.length))
  stem =
    stem
      .replace(/[^A-Za-z0-9_-]+/g, '_')
      .replace(/^_+|_+$/g, '')
      .slice(0, 48) || 'file'
  const rand = crypto.randomBytes(3).toString('hex')
  return `${stem}_${rand}${ext}`
}

export function extractToken(request: Request): string | null {
  const header = request.headers.get('x-bridge-token')
  if (header && header.trim()) return header.trim()
  const q = new URL(request.url).searchParams.get('t')

}

function safeEqualStr(a: string, b: string): boolean {
  const ba = Buffer.from(a)
  const bb = Buffer.from(b)
  if (ba.length !== bb.length || ba.length === 0) return false
  return crypto.timingSafeEqual(ba, bb)
}

/** Возвращает токен при успехе или null */
export async function requireBridgeAuth(request: Request): Promise<string | null> {
  const token = await getBridgeToken()
  const provided = extractToken(request)
  if (!provided || !safeEqualStr(provided, token)) return null
  return token
}

/**
 * Подметание: завершаем зависшие команды и передачи.
 * Вызывается лениво из poll/status/transfers/action.
 */
export async function sweep(): Promise<void> {
  const now = Date.now()
  try {
    // W29-РЕТЕНЦИЯ (раз в час): готовые команды/передачи старше 3 дней
    // удаляем — result-колонки exec/list/read_text раздували БД до 600 МБ.
    if (now - lastRetentionMs > RETENTION_INTERVAL_MS) {
      lastRetentionMs = now
      const cutoff = new Date(now - RETENTION_MS)
      const oldTransfers = await db.bridgeTransfer.findMany({
        where: { status: { in: ['complete', 'failed'] }, updatedAt: { lt: cutoff } },
        select: { stagedName: true, direction: true },
      })
      for (const t of oldTransfers) {
        if (t.direction === 'pc2sb') {
          // staged-файл завершённой pc2sb-передачи больше не нужен
          await fs.promises.rm(stagedPath(t.stagedName), { force: true }).catch(() => {})
        }
      }
      await db.bridgeTransfer.deleteMany({
        where: { status: { in: ['complete', 'failed'] }, updatedAt: { lt: cutoff } },
      })
      await db.bridgeCommand.deleteMany({
        where: { status: { in: ['done', 'failed'] }, doneAt: { lt: cutoff } },
      })
    }
    await db.bridgeCommand.updateMany({
      where: {
        status: 'picked',
        pickedAt: { lt: new Date(now - COMMAND_TIMEOUT_MS) },
        type: { not: 'exec' }, // у exec свой таймаут из payload
      },
      data: {
        status: 'failed',
        error: 'Таймаут выполнения (агент на ПК не ответил)',
        doneAt: new Date(),
      },
    })
    // exec: таймаут = payload.timeoutSec + грейс 120с
    const execPicked = await db.bridgeCommand.findMany({
      where: { status: 'picked', type: 'exec' },
      select: { id: true, payload: true, pickedAt: true },
    })
    for (const c of execPicked) {
      try {
        const p = JSON.parse(c.payload) as { timeoutSec?: number }
        const limitMs = (Number(p.timeoutSec) > 0 ? Number(p.timeoutSec) : 120) * 1000 + 120_000
        if (c.pickedAt && now - c.pickedAt.getTime() > limitMs) {
          await db.bridgeCommand.update({
            where: { id: c.id },
            data: {
              status: 'failed',
              error: 'Таймаут выполнения команды на ПК',
              doneAt: new Date(),
            },
          })
        }
      } catch {
        /* битый payload — пропускаем */
      }
    }
    const stale = await db.bridgeTransfer.findMany({
      where: {
        status: { in: ['pending', 'active'] },
        updatedAt: { lt: new Date(now - TRANSFER_STALL_MS) },
      },
      select: { id: true, direction: true },
    })
    for (const t of stale) {
      await failTransfer(t.id, 'Передача остановлена: нет прогресса 5 минут', t.direction === 'pc2sb')
    }
  } catch {
    /* не роняем запрос из-за sweeper */
  }
}

export function sha256File(p: string): Promise<string> {
  return new Promise((resolve, reject) => {
    try {
      const hash = crypto.createHash('sha256')
      const stream = fs.createReadStream(p)
      stream.on('data', (chunk) => hash.update(chunk))
      stream.on('end', () => resolve(hash.digest('hex')))
      stream.on('error', (e) => reject(e))
    } catch (e) {
      reject(e instanceof Error ? e : new Error(String(e)))
    }
  })
}

/** Валидация env для exec: до 32 пар, ключи/значения до 1024 симв */
export function sanitizeEnv(raw: unknown): Record<string, string> | null {
  if (raw === undefined || raw === null) return {}
  if (typeof raw !== 'object' || Array.isArray(raw)) return null
  const out: Record<string, string> = {}
  const keys = Object.keys(raw as Record<string, unknown>)
  if (keys.length > 32) return null
  for (const k of keys) {
    if (!k || k.length > 256) return null
    const v = String((raw as Record<string, unknown>)[k] ?? '').slice(0, 1024)
    out[k] = v
  }
  return out
}
