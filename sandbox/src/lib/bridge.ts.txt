const RETENTION_MS = 24 * 3600_000
const RETENTION_INTERVAL_MS = 3600_000

export function ensureDirs() {
  fs.mkdirSync(BRIDGE_FILES_DIR, { recursive: true })
  fs.mkdirSync(BRIDGE_TMP_DIR, { recursive: true })
}

export function tmpPath(transferId: string): string {
  return path.join(BRIDGE_TMP_DIR, `${transferId}.part`)
}

export function stagedPath(name: string): string {
  return path.join(BRIDGE_FILES_DIR, name)
}

/** Имя staged-файла: строгий набор символов, без обхода путей */
export function sanitizeStagedName(raw: unknown): string | null {
  const name = String(raw ?? '').trim()
  if (!name || name.length > 100) return null
  if (!/^[A-Za-z0-9][A-Za-z0-9._-]*$/.test(name)) return null
  if (name.includes('..')) return null
  return name
}

/** Из человекочитаемого имени файла (возможно кириллица) делаем безопасное staged-имя */
export function makeStagedName(displayName: string): string {
  const base = path.basename(String(displayName || 'file'))
  const extMatch = path.extname(base).match(/^[A-Za-z0-9.]{0,12}/)
  const ext = extMatch ? extMatch[0] : ''
  let stem = base.slice(0, Math.max(0, base.length - ext.length))
  stem =
    stem
      .replace(/[^A-Za-z0-9_-]+/g, '_')
      .replace(/^_+|_+$/g, '')
      .slice(0, 48) || 'file'
  const rand = crypto.randomBytes(3).toString('hex')
  return `${stem}_${rand}${ext}`
}

// ---------- Настройки ----------

export async function getSetting(key: string): Promise<string | null> {
  const row = await db.bridgeSetting.findUnique({ where: { key } })
  return row?.value ?? null
}

export async function setSetting(key: string, value: string): Promise<void> {
  await db.bridgeSetting.upsert({
    where: { key },
    create: { key, value },
    update: { value },
  })
}

export async function getBridgeToken(): Promise<string> {
  const row = await db.bridgeSetting.findUnique({ where: { key: 'token' } })
  if (row) return row.value
  const token = crypto.randomBytes(24).toString('hex')
  try {
    await db.bridgeSetting.create({ data: { key: 'token', value: token } })
    return token
  } catch {
    const existing = await db.bridgeSetting.findUnique({ where: { key: 'token' } })
    return existing?.value ?? token
  }
}

export async function rotateBridgeToken(): Promise<string> {
  const token = crypto.randomBytes(24).toString('hex')
  await setSetting('token', token)
  return token
}

// ---------- Авторизация ----------

export function extractToken(request: Request): string | null {
  const header = request.headers.get('x-bridge-token')
  if (header && header.trim()) return header.trim()
  const q = new URL(request.url).searchParams.get('t')
  if (q && q.trim()) return q.trim()
  return null
}

function safeEqualStr(a: string, b: string): boolean {
  const ba = Buffer.from(a)
  const bb = Buffer.from(b)
  if (ba.length !== bb.length || ba.length === 0) return false
  return crypto.timingSafeEqual(ba, bb)
}

/** Возвращает токен при успехе или null */
export async function requireBridgeAuth(request: Request): Promise<string | null> {
  const token = await getBridgeToken()
  const provided = extractToken(request)
  if (!provided || !safeEqualStr(provided, token)) return null
  return token
}

export async function touchClient(clientId: string, info: unknown): Promise<void> {
  const now = new Date()
  await setSetting('lastSeenAt', now.toISOString())
  if (clientId) await setSetting('clientId', clientId)
  if (info !== undefined && info !== null) {
    await setSetting('clientInfo', JSON.stringify(info))
  }
}

async function failTransfer(id: string, error: string, cleanTmp: boolean): Promise<void> {
  try {
    await db.bridgeTransfer.update({ where: { id }, data: { status: 'failed', error } })
  } catch {
    /* передача могла быть уже удалена */
  }
  if (cleanTmp) {
    await fs.promises.rm(tmpPath(id), { force: true }).catch(() => {})
  }
}

/**
 * Подметание: завершаем зависшие команды и передачи.
 * Вызывается лениво из poll/status/transfers/action.
 */
export async function sweep(): Promise<void> {
  const now = Date.now()
  try {
    // W29-РЕТЕНЦИЯ (раз в час): готовые команды/передачи старше 3 дней
    // удаляем — result-колонки exec/list/read_text раздували БД до 600 МБ.
    if (now - lastRetentionMs > RETENTION_INTERVAL_MS) {
      lastRetentionMs = now
      const cutoff = new Date(now - RETENTION_MS)
      const oldTransfers = await db.bridgeTransfer.findMany({
        where: { status: { in: ['complete', 'failed'] }, updatedAt: { lt: cutoff } },
        select: { stagedName: true, direction: true },
      })
      for (const t of oldTransfers) {
        if (t.direction === 'pc2sb') {
          // staged-файл завершённой pc2sb-передачи больше не нужен
          await fs.promises.rm(stagedPath(t.stagedName), { force: true }).catch(() => {})
        }
      }
      await db.bridgeTransfer.deleteMany({
        where: { status: { in: ['complete', 'failed'] }, updatedAt: { lt: cutoff } },
      })
      await db.bridgeCommand.deleteMany({
        where: { status: { in: ['done', 'failed'] }, doneAt: { lt: cutoff } },
      })
    }
    await db.bridgeCommand.updateMany({
      where: {
        status: 'picked',
        pickedAt: { lt: new Date(now - COMMAND_TIMEOUT_MS) },
        type: { not: 'exec' }, // у exec свой таймаут из payload
      },
      data: {
        status: 'failed',
        error: 'Таймаут выполнения (агент на ПК не ответил)',
        doneAt: new Date(),
      },
    })
    // exec: таймаут = payload.timeoutSec + грейс 120с
    const execPicked = await db.bridgeCommand.findMany({
      where: { status: 'picked', type: 'exec' },
      select: { id: true, payload: true, pickedAt: true },
    })
    for (const c of execPicked) {
      try {
        const p = JSON.parse(c.payload) as { timeoutSec?: number }
        const limitMs = (Number(p.timeoutSec) > 0 ? Number(p.timeoutSec) : 120) * 1000 + 120_000
        if (c.pickedAt && now - c.pickedAt.getTime() > limitMs) {
          await db.bridgeCommand.update({
            where: { id: c.id },
            data: {
              status: 'failed',
              error: 'Таймаут выполнения команды на ПК',
              doneAt: new Date(),
            },
          })
        }
      } catch {
        /* битый payload — пропускаем */
      }
    }
    const stale = await db.bridgeTransfer.findMany({
      where: {
        status: { in: ['pending', 'active'] },
        updatedAt: { lt: new Date(now - TRANSFER_STALL_MS) },
      },
      select: { id: true, direction: true },
    })
    for (const t of stale) {
      await failTransfer(t.id, 'Передача остановлена: нет прогресса 5 минут', t.direction === 'pc2sb')
    }
  } catch {
    /* не роняем запрос из-за sweeper */
  }
}

// ---------- Файлы ----------

export function sha256File(p: string): Promise<string> {
  return new Promise((resolve, reject) => {
    try {
      const hash = crypto.createHash('sha256')
      const stream = fs.createReadStream(p)
      stream.on('data', (chunk) => hash.update(chunk))
      stream.on('end', () => resolve(hash.digest('hex')))
      stream.on('error', (e) => reject(e))
    } catch (e) {
      reject(e instanceof Error ? e : new Error(String(e)))
    }
  })
}

export async function appendChunk(transferId: string, buf: Buffer): Promise<void> {
  await fs.promises.appendFile(tmpPath(transferId), buf)
}

export function sleep(ms: number): Promise<void> {
  return new Promise((resolve) => setTimeout(resolve, ms))
}

/** Ограничение строки-пути из запроса */
export function clampPath(raw: unknown, maxLen = 512): string {
  return String(raw ?? '').slice(0, maxLen)
}

/** Ограничение команды */
export function clampCmd(raw: unknown): string {
  return String(raw ?? '').slice(0, EXEC_MAX_CMD_CHARS)
}

/** Валидация env для exec: до 32 пар, ключи/значения до 1024 симв */
export function sanitizeEnv(raw: unknown): Record<string, string> | null {
  if (raw === undefined || raw === null) return {}
  if (typeof raw !== 'object' || Array.isArray(raw)) return null
  const out: Record<string, string> = {}
  const keys = Object.keys(raw as Record<string, unknown>)
  if (keys.length > 32) return null
  for (const k of keys) {
    if (!k || k.length > 256) return null
    const v = String((raw as Record<string, unknown>)[k] ?? '').slice(0, 1024)
    out[k] = v
  }
  return out
}
