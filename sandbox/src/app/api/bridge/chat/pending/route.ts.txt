import { db } from '@/lib/db'
import { requireBridgeAuth, jok, jerr } from '@/lib/bridge'

export const dynamic = 'force-dynamic'

/**
 * GET /api/bridge/chat/pending — необработанные сообщения пользователя
 * (их получает ИИ-агент песочницы в цикле автономной работы).
 */
export async function GET(request: Request) {
  const auth = await requireBridgeAuth(request)
  if (!auth) return jerr('Неверный токен моста', 401)

  const rows = await db.bridgeChatMessage.findMany({
    where: { role: 'user', status: 'new' },
    orderBy: { createdAt: 'asc' },
    take: 100,
  })

  return jok({
    messages: rows.map((m) => ({
      id: m.id,
      role: m.role,
      text: m.text,
      createdAt: m.createdAt.toISOString(),
    })),
    count: rows.length,
  })
}
