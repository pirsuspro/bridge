import { db } from '@/lib/db'
import { requireBridgeAuth, jerr, jok, sweep } from '@/lib/bridge'

export const dynamic = 'force-dynamic'

/**
 * GET /api/bridge/transfers — последние 25 передач (новые сверху).
 */
export async function GET(request: Request) {
  const token = await requireBridgeAuth(request)
  if (!token) return jerr('Неверный токен моста', 401)

  await sweep()
  const transfers = await db.bridgeTransfer.findMany({
    orderBy: { createdAt: 'desc' },
    take: 25,
  })

  return jok({
    transfers: transfers.map((t) => ({
      id: t.id,
      direction: t.direction,
      path: t.path,
      stagedName: t.stagedName,
      size: t.size,
      bytesDone: t.bytesDone,
      sha256: t.sha256,
      status: t.status,
      error: t.error,
      createdAt: t.createdAt.toISOString(),
      updatedAt: t.updatedAt.toISOString(),
    })),
  })
}
